local autoQTEEnabled = false
local qteConnections = {}

-- Function to press a key
local function pressKey(key)
    print("Attempting to press key: " .. key)
    
    -- Handle special keys
    local keyMap = {
        ["Space"] = Enum.KeyCode.Space,
        ["Enter"] = Enum.KeyCode.Return,
        ["Shift"] = Enum.KeyCode.LeftShift,
        ["Control"] = Enum.KeyCode.LeftControl,
        ["Alt"] = Enum.KeyCode.LeftAlt,
        ["Tab"] = Enum.KeyCode.Tab,
        ["Backspace"] = Enum.KeyCode.Backspace,
        ["Delete"] = Enum.KeyCode.Delete,
        ["Escape"] = Enum.KeyCode.Escape,
        ["Up"] = Enum.KeyCode.Up,
        ["Down"] = Enum.KeyCode.Down,
        ["Left"] = Enum.KeyCode.Left,
        ["Right"] = Enum.KeyCode.Right,
    }
    
    local keyCode
    
    -- Check if it's a special key
    if keyMap[key] then
        keyCode = keyMap[key]
    else
        -- Try to get the key code directly
        keyCode = Enum.KeyCode[key]
        
        -- If that fails, try to find it by name
        if not keyCode then
            for _, enum in ipairs(Enum.KeyCode:GetEnumItems()) do
                if enum.Name == key then
                    keyCode = enum
                    break
                end
            end
        end
    end
    
    if keyCode then
        print("Pressing key: " .. tostring(keyCode))
        VirtualInputManager:SendKeyEvent(true, keyCode, false, game)
        task.wait(0.05)
        VirtualInputManager:SendKeyEvent(false, keyCode, false, game)
        print("Pressed key: " .. key)
    else
        print("Could not find key: " .. key)
    end
end

-- Function to monitor a specific QTE
local function monitorQTE(qtePath, qteName)
    local qteText = game.Players.LocalPlayer.PlayerGui:FindFirstChild("Ingame")
    if qteText then
        qteText = qteText:FindFirstChild("MechanicsFrame")
        if qteText then
            qteText = qteText:FindFirstChild(qtePath)
            if qteText then
                qteText = qteText:FindFirstChild("ImageButton")
                if qteText then
                    qteText = qteText:FindFirstChild("PC")
                end
            end
        end
    end
    
    if not qteText then
        warn("QTE UI not found: " .. qteName)
        return false
    end
    
    print("QTE monitor started for: " .. qteName)
    
    -- Property change listener
    local connection = qteText:GetPropertyChangedSignal("Text"):Connect(function()
        if not autoQTEEnabled then return end
        local currentText = qteText.Text
        if currentText ~= "Hi" and currentText ~= "" then
            print("[" .. qteName .. "] QTE Key detected: " .. currentText)
            pressKey(currentText)
            task.wait(0.1)
            qteText.Text = "Hi"
        end
    end)
    table.insert(qteConnections, connection)
    
    -- Polling loop as backup
    local thread = task.spawn(function()
        while autoQTEEnabled do
            task.wait(0.1)
            if not autoQTEEnabled then break end
            local currentText = qteText.Text
            if currentText ~= "Hi" and currentText ~= "" then
                print("[" .. qteName .. "] QTE Key detected (poll): " .. currentText)
                pressKey(currentText)
                task.wait(0.1)
                qteText.Text = "Hi"
            end
        end
    end)
    table.insert(qtePollingThreads, thread)
    
    return true
end

-- Toggle function
local function toggleQTE(Value)
    autoQTEEnabled = Value
    
    if Value then
        -- Clean up old connections if any
        for _, conn in ipairs(qteConnections) do
            conn:Disconnect()
        end
        qteConnections = {}
        
        for _, thread in ipairs(qtePollingThreads) do
            task.cancel(thread)
        end
        qtePollingThreads = {}
        
        -- Start monitoring both QTEs
        monitorQTE("QTEToma", "QTEToma")
        monitorQTE("QTE", "QTE")
    else
        -- Clean up connections
        for _, conn in ipairs(qteConnections) do
            conn:Disconnect()
        end
        qteConnections = {}
        
        for _, thread in ipairs(qtePollingThreads) do
            task.cancel(thread)
        end
        qtePollingThreads = {}
    end
end

local autoDodgeChokeEnabled = false
local soundConnections = {}
local dodgeChokeCooldown = 0
local TARGET_AUDIO_ID = "16214979367"
local DODGE_COOLDOWN = 0.65
local DODGE_DELAY = 0.3 -- 0.3 second delay before dodging

-- Function to check distance to CHAIN
local function isNearChain()
    local player = game.Players.LocalPlayer
    local character = player.Character
    if not character then return false end
    
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    
    local chain = workspace:FindFirstChild("Misc")
    if chain then
        chain = chain:FindFirstChild("AI")
        if chain then
            chain = chain:FindFirstChild("CHAIN")
        end
    end
    
    if not chain then return false end
    
    local chainHrp = chain:FindFirstChild("HumanoidRootPart")
    if not chainHrp then return false end
    
    local distance = (hrp.Position - chainHrp.Position).Magnitude
    return distance <= 15
end

-- Function to press Q (dodge) with delay
local function pressQKeyWithDelay()
    task.wait(DODGE_DELAY) -- Wait 0.3 seconds before dodging
    VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.Q, false, game)
    task.wait(0.05)
    VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.Q, false, game)
end

-- Process sound for dodge choke
local function hookSound(sound)
    if not sound:IsA("Sound") then return end
    
    local function checkAndTrigger()
        if not autoDodgeChokeEnabled then return end
        
        local soundId = tostring(sound.SoundId)
        if soundId:find(TARGET_AUDIO_ID) then
            local now = os.clock()
            if (now - dodgeChokeCooldown) >= DODGE_COOLDOWN then
                -- Check if player is near CHAIN (within 15 studs)
                if isNearChain() then
                    dodgeChokeCooldown = now
                    task.spawn(pressQKeyWithDelay) -- Dodge with 0.3s delay
                    print("Auto Dodge Choke triggered! (0.3s delay)")
                else
                    local chain = workspace:FindFirstChild("Misc")
                    if chain then
                        chain = chain:FindFirstChild("AI")
                        if chain then
                            chain = chain:FindFirstChild("CHAIN")
                        end
                    end
                    if chain and chain:FindFirstChild("HumanoidRootPart") then
                        local player = game.Players.LocalPlayer
                        local character = player.Character
                        if character and character:FindFirstChild("HumanoidRootPart") then
                            local distance = (character.HumanoidRootPart.Position - chain.HumanoidRootPart.Position).Magnitude
                            print("CHAIN too far (" .. math.floor(distance) .. " studs) - not dodging")
                        end
                    end
                end
            end
        end
    end
    
    -- Connect to Played event
    sound.Played:Connect(checkAndTrigger)
    
    -- Also check when IsPlaying changes
    sound:GetPropertyChangedSignal("IsPlaying"):Connect(function()
        if sound.IsPlaying then
            checkAndTrigger()
        end
    end)
    
    table.insert(soundConnections, sound)
end

-- Setup sound hooks
local function setupDodgeChoke()
    -- Hook all existing sounds
    for _, desc in pairs(game:GetDescendants()) do
        pcall(function()
            hookSound(desc)
        end)
    end
    
    -- Hook new sounds as they're added
    game.DescendantAdded:Connect(function(desc)
        pcall(function()
            hookSound(desc)
        end)
    end)
end

-- Toggle function for Auto Dodge Choke
local function toggleDodgeChoke(Value)
    autoDodgeChokeEnabled = Value
    
    if Value then
        setupDodgeChoke()
        print("Auto Dodge Choke Enabled! (0.3s delay)")
    else
        print("Auto Dodge Choke Disabled!")
    end
end

-- ===== SCRAP ESP (WITH DEATH CLEANUP) =====
local scrapESPEnabled = false
local scrapComponents = {}
local scrapScreenGui = nil
local scrapBin = nil
local scrapScanConnection = nil
local scrapDeathConnection = nil

-- Create Bin class for cleanup
local function createBin()
    local bin = {}
    local items = {}
    
    function bin:add(item)
        table.insert(items, item)
        return self
    end
    
    function bin:destroy()
        for _, item in ipairs(items) do
            if type(item) == "function" then
                pcall(item)
            elseif typeof(item) == "RBXScriptConnection" then
                pcall(function() item:Disconnect() end)
            elseif type(item) == "thread" then
                pcall(function() task.cancel(item) end)
            elseif item and type(item) == "table" and item.destroy then
                pcall(function() item:destroy() end)
            elseif item and item.Destroy then
                pcall(function() item:Destroy() end)
            end
        end
        items = {}
    end
    
    return bin
end

-- Function to clear all scrap ESP
local function clearScrapESP()
    if scrapBin then
        pcall(function() scrapBin:destroy() end)
        scrapBin = nil
    end
    
    if scrapScreenGui then
        pcall(function() scrapScreenGui:Destroy() end)
        scrapScreenGui = nil
    end
    
    scrapComponents = {}
end

-- Function to format numbers
local function formatNum(num, format)
    local formatted = string.format(`%.{format}f`, num)
    return formatted
end

-- LootableComponent class for each scrap item
local LootableComponent = {}
LootableComponent.__index = LootableComponent

function LootableComponent.new(scrap, pivot)
    local self = setmetatable({}, LootableComponent)
    self.object = scrap
    self.pivotPos = pivot
    self.bin = createBin()
    self.isAvailable = true
    
    -- Check if scrap has Values folder with Available attribute
    local values = scrap:FindFirstChild("Values")
    if values then
        self.isAvailable = values:GetAttribute("Available") ~= false
    end
    
    -- Create UI elements
    self.container = Instance.new("Frame")
    self.nameLabel = Instance.new("TextLabel")
    
    self.container.Visible = false
    self.container.AnchorPoint = Vector2.new(0.5, 0)
    self.container.BackgroundTransparency = 1
    self.container.Size = UDim2.new(0, 100, 0, 50)
    
    self.nameLabel.BackgroundTransparency = 1
    self.nameLabel.Font = Enum.Font.Nunito
    self.nameLabel.Text = "Scrap"
    self.nameLabel.TextColor3 = Color3.new(0, 1, 0)
    self.nameLabel.TextSize = 15
    self.nameLabel.TextStrokeTransparency = 0.5
    self.nameLabel.Size = UDim2.new(1, 0, 0, 14)
    
    self.nameLabel.Parent = self.container
    
    -- Add to ScreenGui
    if scrapScreenGui then
        self.container.Parent = scrapScreenGui
    end
    
    -- Track availability changes
    if values then
        local conn = values:GetAttributeChangedSignal("Available"):Connect(function()
            self.isAvailable = values:GetAttribute("Available")
        end)
        self.bin:add(conn)
    end
    
    -- Render loop
    local renderConn
    renderConn = game:GetService("RunService").RenderStepped:Connect(function()
        self:render()
    end)
    self.bin:add(renderConn)
    
    -- Track object death
    local deathConn = scrap.AncestryChanged:Connect(function()
        if not scrap or not scrap.Parent then
            self:destroy()
        end
    end)
    self.bin:add(deathConn)
    
    return self
end

function LootableComponent:render()
    local camera = workspace.CurrentCamera
    local player = game.Players.LocalPlayer
    local character = player.Character
    
    if not camera or not character then
        self.container.Visible = false
        return
    end
    
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then
        self.container.Visible = false
        return
    end
    
    if not self.object or not self.object.Parent then
        self:destroy()
        return
    end
    
    local position, visible = camera:WorldToViewportPoint(self.pivotPos.Position)
    
    if visible and self.isAvailable then
        local scale = 1 / (position.Z * math.tan(math.rad(camera.FieldOfView * 0.5)) * 2) * 1000
        local width, height = math.floor(1 * scale), math.floor(3 * scale)
        local x, y = math.floor(position.X), math.floor(position.Y)
        local xPosition, yPosition = math.floor(x - width * 0.5), math.floor((y - height * 0.5) + (0.5 * scale))
        
        self.container.Visible = true
        self.container.Position = UDim2.new(0, xPosition, 0, yPosition)
        
        local distance = (hrp.Position - self.pivotPos.Position).Magnitude
        self.nameLabel.Text = string.format("Scrap [%.1f]", distance)
    else
        self.container.Visible = false
    end
end

function LootableComponent:destroy()
    if self.bin then
        pcall(function() self.bin:destroy() end)
        self.bin = nil
    end
    if self.container then
        pcall(function() self.container:Destroy() end)
        self.container = nil
    end
    self.object = nil
end

-- Function to setup scrap ESP
local function setupScrapESP()
    clearScrapESP()
    
    if not scrapESPEnabled then return end
    
    -- Create ScreenGui
    scrapScreenGui = Instance.new("ScreenGui")
    scrapScreenGui.DisplayOrder = 10
    scrapScreenGui.IgnoreGuiInset = true
    scrapScreenGui.Parent = game:GetService("CoreGui")
    scrapScreenGui.Name = "ScrapESP"
    
    scrapBin = createBin()
    
    -- Find scrap folder
    local scrapFolder = workspace:FindFirstChild("Misc")
    if scrapFolder then
        scrapFolder = scrapFolder:FindFirstChild("Zones")
        if scrapFolder then
            scrapFolder = scrapFolder:FindFirstChild("LootingItems")
            if scrapFolder then
                scrapFolder = scrapFolder:FindFirstChild("Scrap")
            end
        end
    end
    
    if not scrapFolder then return end
    
    -- Create components for existing scrap
    for _, scrap in ipairs(scrapFolder:GetChildren()) do
        if scrap:IsA("Model") then
            local component = LootableComponent.new(scrap, scrap:GetPivot())
            table.insert(scrapComponents, component)
        end
    end
    
    -- Watch for new scrap
    local newConn = scrapFolder.ChildAdded:Connect(function(child)
        if child:IsA("Model") then
            local component = LootableComponent.new(child, child:GetPivot())
            table.insert(scrapComponents, component)
        end
    end)
    scrapBin:add(newConn)
    
    -- Watch for scrap removal
    local removedConn = scrapFolder.ChildRemoved:Connect(function(child)
        for i, comp in ipairs(scrapComponents) do
            if comp.object == child then
                pcall(function() comp:destroy() end)
                table.remove(scrapComponents, i)
                break
            end
        end
    end)
    scrapBin:add(removedConn)
end

-- Function to handle player death/respawn
local function setupDeathHandling()
    local player = game.Players.LocalPlayer
    
    -- When character dies, clear ESP
    local deathConn = player.CharacterAdded:Connect(function(character)
        -- Clear everything when respawning
        clearScrapESP()
        
        -- Wait a moment then re-setup if enabled
        task.wait(0.5)
        if scrapESPEnabled then
            setupScrapESP()
        end
    end)
    
    scrapDeathConnection = deathConn
end

-- Toggle for Scrap ESP
local function toggleScrapESP(Value)
    scrapESPEnabled = Value
    
    if scrapScanConnection then
        pcall(function() scrapScanConnection:Disconnect() end)
        scrapScanConnection = nil
    end
    
    if Value then
        setupScrapESP()
        setupDeathHandling()
        
        -- Periodic scan for new scrap
        scrapScanConnection = game:GetService("RunService").Heartbeat:Connect(function()
            if scrapESPEnabled then
                -- Check if scrap folder exists and has new items
                local scrapFolder = workspace:FindFirstChild("Misc")
                if scrapFolder then
                    scrapFolder = scrapFolder:FindFirstChild("Zones")
                    if scrapFolder then
                        scrapFolder = scrapFolder:FindFirstChild("LootingItems")
                        if scrapFolder then
                            scrapFolder = scrapFolder:FindFirstChild("Scrap")
                        end
                    end
                end
                
                if not scrapFolder then
                    clearScrapESP()
                    return
                end
                
                -- Check for any scrap items without components
                local existingObjects = {}
                for _, comp in ipairs(scrapComponents) do
                    if comp.object then
                        existingObjects[comp.object] = true
                    end
                end
                
                for _, scrap in ipairs(scrapFolder:GetChildren()) do
                    if scrap:IsA("Model") and not existingObjects[scrap] then
                        local component = LootableComponent.new(scrap, scrap:GetPivot())
                        table.insert(scrapComponents, component)
                    end
                end
            else
                if scrapScanConnection then
                    pcall(function() scrapScanConnection:Disconnect() end)
                    scrapScanConnection = nil
                end
            end
        end)
    else
        clearScrapESP()
        if scrapDeathConnection then
            pcall(function() scrapDeathConnection:Disconnect() end)
            scrapDeathConnection = nil
        end
    end
end

local Toggle = Tab:CreateToggle({
   Name = "Scrap ESP",
   CurrentValue = false,
   Flag = "ScrapESP",
   Callback = function(Value)
      toggleScrapESP(Value)
   end,

-- Function to find a location by path
local function findLocation(path)
    local current = workspace
    for _, part in ipairs(path) do
        current = current:FindFirstChild(part)
        if not current then
            return nil
        end
    end
    return current
end

-- Function to clear location ESP
local function clearLocationESP()
    if locationESPBin then
        pcall(function() locationESPBin:destroy() end)
        locationESPBin = nil
    end
    
    for _, data in ipairs(locationESPObjects) do
        if data.billboard then
            pcall(function() data.billboard:Destroy() end)
        end
        if data.connection then
            pcall(function() data.connection:Disconnect() end)
        end
    end
    locationESPObjects = {}
end

-- Function to setup location ESP
local function setupLocationESP()
    clearLocationESP()
    
    if not locationESPEnabled then return end
    
    locationESPBin = createBin()
    
    for _, locationData in ipairs(LOCATIONS) do
        local location = findLocation(locationData.path)
        if location then
            -- Find a part to attach the billboard to
            local attachPart = location:FindFirstChild("HumanoidRootPart") or location:FindFirstChild("Head") or location:FindFirstChildOfClass("BasePart")
            if not attachPart then
                attachPart = location
            end
            
            -- Create BillboardGui
            local billboard = Instance.new("BillboardGui")
            billboard.Name = "LocationESP_" .. locationData.name
            billboard.Size = UDim2.new(0, 200, 0, 40)
            billboard.StudsOffset = Vector3.new(0, 5, 0)
            billboard.AlwaysOnTop = true
            billboard.Parent = attachPart
            
            local textLabel = Instance.new("TextLabel")
            textLabel.Size = UDim2.new(1, 0, 1, 0)
            textLabel.BackgroundTransparency = 1
            textLabel.Text = locationData.name
            textLabel.TextColor3 = Color3.fromRGB(0, 255, 255)
            textLabel.TextStrokeTransparency = 0
            textLabel.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
            textLabel.TextScaled = true
            textLabel.Font = Enum.Font.GothamBold
            textLabel.Parent = billboard
            
            -- Watch for location being destroyed
            local conn = location.AncestryChanged:Connect(function()
                if not location or not location.Parent then
                    pcall(function() billboard:Destroy() end)
                    if conn then pcall(function() conn:Disconnect() end) end
                end
            end)
            
            table.insert(locationESPObjects, {
                name = locationData.name,
                location = location,
                billboard = billboard,
                textLabel = textLabel,
                connection = conn
            })
            
            locationESPBin:add(conn)
        end
    end
end

                    pcall(function() locationESPScanConnection:Disconnect() end)

                end
            end
        end)

    end
