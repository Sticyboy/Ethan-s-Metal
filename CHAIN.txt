-- Key binding functions
local function getKeyName(inputType)
    if inputType == Enum.UserInputType.MouseButton1 then
        return "Left Click"
    elseif inputType == Enum.UserInputType.MouseButton2 then
        return "Right Click"
    elseif inputType == Enum.UserInputType.MouseButton3 then
        return "Middle Click"
    elseif inputType.Name:match("Keyboard") then
        return inputType.Name:gsub("Keyboard", "")
    else
        return tostring(inputType.Name)
    end
end

-- FOV Circle Visual
local function createFOVCircle()

local function isHoldingItem()
    if not RequireHoldingTool then
        return true
    end
    
    if not LocalPlayer.Character then
        return false
    end
    
    local tool = LocalPlayer.Character:FindFirstChildOfClass("Tool")
    if tool then
        return true
    end
    
    return false
end

local function isAlive(character)
    if not character then return false end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if humanoid then
        return humanoid.Health > 0
    end
    return true
end

local function wallCheck(targetPosition)
    if not WallCheckEnabled then
        return false
    end
    
    local origin = Camera.CFrame.Position
    local direction = (targetPosition - origin).Unit
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Blacklist
    raycastParams.FilterDescendantsInstances = {LocalPlayer.Character, Camera}
    
    local raycastResult = Workspace:Raycast(origin, direction * 1000, raycastParams)
    
    if raycastResult then
        local distanceToTarget = (targetPosition - origin).Magnitude
        local distanceToHit = (raycastResult.Position - origin).Magnitude
        return distanceToHit < distanceToTarget - 1
    end
    
    return false
end

local function isMouseInFOV()
    local mouseLocation = UserInputService:GetMouseLocation()
    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local distanceFromCenter = (mouseLocation - screenCenter).Magnitude
    return distanceFromCenter <= AimbotFOV
end

local function isTargetInFOV(targetPart)
    if not targetPart then
        return false
    end
    
    if AimbotFOV >= 180 then
        return true
    end
    
    local targetPosition, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
    
    if not onScreen then
        return false
    end
    
    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local distance = (Vector2.new(targetPosition.X, targetPosition.Y) - screenCenter).Magnitude
    
    return distance <= AimbotFOV
end

-- Get CHAIN target
local function getChainTargetPart()
    local chain = workspace:FindFirstChild("Misc")
    if chain then
        chain = chain:FindFirstChild("AI")
        if chain then
            chain = chain:FindFirstChild("CHAIN")
        end
    end
    
    if not chain then
        return nil
    end
    
    if not isAlive(chain) then
        return nil
    end
    
    local head = chain:FindFirstChild("Head")
    if head then
        return head
    end
    
    local humanoidRootPart = chain:FindFirstChild("HumanoidRootPart")
    if humanoidRootPart then
        return humanoidRootPart
    end
    
    local torso = chain:FindFirstChild("Torso") or chain:FindFirstChild("UpperTorso")
    return torso
end

local function getChainTarget()
    local targetPart = getChainTargetPart()
    if targetPart then
        if isTargetInFOV(targetPart) then
            if not wallCheck(targetPart.Position) then
                return targetPart
            end
        end
    end
    return nil
end

-- AIM AT TARGET WITH SMOOTHNESS (from your script)
local function aimAtTarget(targetPart)
    if not targetPart then return end
    
    local targetPosition = targetPart.Position
    local lookAt = CFrame.lookAt(Camera.CFrame.Position, targetPosition)
    
    -- Using the smoothness from your script
    Camera.CFrame = Camera.CFrame:Lerp(lookAt, AimbotSmoothness)
end

-- Check if chain exists and is valid
local function isChainValid()
    local chain = workspace:FindFirstChild("Misc")
    if chain then
        chain = chain:FindFirstChild("AI")
        if chain then
            chain = chain:FindFirstChild("CHAIN")
        end
    end
    return chain ~= nil and isAlive(chain)
end

local function startAimbot()
    if aimbotConnection then
        aimbotConnection:Disconnect()
    end
    
    aimbotConnection = RunService.RenderStepped:Connect(function()
        if AimbotEnabled and UserInputService:GetFocusedTextBox() == nil then
            local isAimbotKeyPressed = false
            
            if AimbotKey == Enum.UserInputType.MouseButton1 or AimbotKey == Enum.UserInputType.MouseButton2 or AimbotKey == Enum.UserInputType.MouseButton3 then
                isAimbotKeyPressed = UserInputService:IsMouseButtonPressed(AimbotKey)
            else
                isAimbotKeyPressed = UserInputService:IsKeyDown(Enum.KeyCode[AimbotKey.Name])
            end
            
            local shouldAim = false
            if RequireHoldingTool then
                shouldAim = isHoldingItem() and isMouseInFOV()
            else
                shouldAim = isMouseInFOV()
            end
            
            if isAimbotKeyPressed and not prevKeyPressed then
                WaitingForRelease = false
            end
            
            if not isAimbotKeyPressed then
                WaitingForRelease = false
                if isAiming then
                    currentAimTarget = nil
                    isAiming = false
                end
            end
            
            prevKeyPressed = isAimbotKeyPressed
            
            if isAimbotKeyPressed and shouldAim then
                local chainTarget = getChainTarget()
                
                if not isAiming and not WaitingForRelease then
                    if chainTarget then
                        currentAimTarget = chainTarget
                        isAiming = true
                    end
                end
                
                if currentAimTarget and (not currentAimTarget.Parent or not isAlive(currentAimTarget.Parent) or wallCheck(currentAimTarget.Position)) then
                    if AutoRetarget then
                        if chainTarget then
                            currentAimTarget = chainTarget
                            isAiming = true
                        else
                            currentAimTarget = nil
                            isAiming = false
                        end
                    else
                        currentAimTarget = nil
                        isAiming = false
                        WaitingForRelease = true
                    end
                end
                
                if currentAimTarget and currentAimTarget.Parent and isAlive(currentAimTarget.Parent) then
                    if isTargetInFOV(currentAimTarget) and not wallCheck(currentAimTarget.Position) then
                        aimAtTarget(currentAimTarget)
                    else
                        if AutoRetarget then
                            if chainTarget then
                                currentAimTarget = chainTarget
                                isAiming = true
                            else
                                currentAimTarget = nil
                                isAiming = false
                            end
                        else
                            if currentAimTarget then
                                currentAimTarget = nil
                                isAiming = false
                                WaitingForRelease = true
                            end
                        end
                    end
                else
                    if AutoRetarget then
                        if chainTarget then
                            currentAimTarget = chainTarget
                            isAiming = true
                        else
                            currentAimTarget = nil
                            isAiming = false
                        end
                    else
                        if currentAimTarget then
                            currentAimTarget = nil
                            isAiming = false
                            WaitingForRelease = true
                        end
                    end
                end
            else
                if isAiming then
                    currentAimTarget = nil
                    isAiming = false
                end
            end
        else
            currentAimTarget = nil
            isAiming = false
            WaitingForRelease = false
            prevKeyPressed = false
        end
    end)
end

local function stopAimbot()
    if aimbotConnection then
        aimbotConnection:Disconnect()
        aimbotConnection = nil
    end
    currentAimTarget = nil
    isAiming = false
    WaitingForRelease = false
    prevKeyPressed = false
end

local autoQTEEnabled = false

local function pressKey(key)
    print("Attempting to press key: " .. key)
    local keyMap = {
        ["Space"] = Enum.KeyCode.Space,
        ["Enter"] = Enum.KeyCode.Return,
        ["Shift"] = Enum.KeyCode.LeftShift,
        ["Control"] = Enum.KeyCode.LeftControl,
        ["Alt"] = Enum.KeyCode.LeftAlt,
        ["Tab"] = Enum.KeyCode.Tab,
        ["Backspace"] = Enum.KeyCode.Backspace,
        ["Delete"] = Enum.KeyCode.Delete,
        ["Escape"] = Enum.KeyCode.Escape,
        ["Up"] = Enum.KeyCode.Up,
        ["Down"] = Enum.KeyCode.Down,
        ["Left"] = Enum.KeyCode.Left,
        ["Right"] = Enum.KeyCode.Right,
    }
    local keyCode
    if keyMap[key] then
        keyCode = keyMap[key]
    else
        keyCode = Enum.KeyCode[key]
        if not keyCode then
            for _, enum in ipairs(Enum.KeyCode:GetEnumItems()) do
                if enum.Name == key then
                    keyCode = enum
                    break
                end
            end
        end
    end
    if keyCode then
        VirtualInputManager:SendKeyEvent(true, keyCode, false, game)
        task.wait(0.05)
        VirtualInputManager:SendKeyEvent(false, keyCode, false, game)
        print("Pressed key: " .. key)
    else
        print("Could not find key: " .. key)
    end
end

local function monitorQTE(qtePath, qteName)
    local qteText = game.Players.LocalPlayer.PlayerGui:FindFirstChild("Ingame")
    if qteText then
        qteText = qteText:FindFirstChild("MechanicsFrame")

            qteText = qteText:FindFirstChild(qtePath)
            if qteText then
                qteText = qteText:FindFirstChild("ImageButton")
                if qteText then
                    qteText = qteText:FindFirstChild("PC")
                end
            end
        end
    end
    if not qteText then
        warn("QTE UI not found: " .. qteName)
        return false
    end
    print("QTE monitor started for: " .. qteName)
    local connection = qteText:GetPropertyChangedSignal("Text"):Connect(function()
        if not autoQTEEnabled then return end
        local currentText = qteText.Text
        if currentText ~= "Hi" and currentText ~= "" then
            pressKey(currentText)
            task.wait(0.1)
            qteText.Text = "Hi"
        end
    end)
    table.insert(qteConnections, connection)
    local thread = task.spawn(function()
        while autoQTEEnabled do
            task.wait(0.1)
            if not autoQTEEnabled then break end
            local currentText = qteText.Text
            if currentText ~= "Hi" and currentText ~= "" then
                pressKey(currentText)
                task.wait(0.1)
                qteText.Text = "Hi"

local function createBin()
    local bin = {}
    local items = {}
    function bin:add(item) table.insert(items, item) return self end
    function bin:destroy()
        for _, item in ipairs(items) do
            if type(item) == "function" then pcall(item)
            elseif typeof(item) == "RBXScriptConnection" then pcall(function() item:Disconnect() end)
            elseif type(item) == "thread" then pcall(function() task.cancel(item) end)
            elseif item and type(item) == "table" and item.destroy then pcall(function() item:destroy() end)
            elseif item and item.Destroy then pcall(function() item:Destroy() end)
            end
        end
        items = {}
    end
    return bin
end

local function clearScrapESP()
    if scrapBin then pcall(function() scrapBin:destroy() end) scrapBin = nil end
    if scrapScreenGui then pcall(function() scrapScreenGui:Destroy() end) scrapScreenGui = nil end
    scrapComponents = {}
end

local function formatNum(num, format)
    return string.format("%." .. format .. "f", num)
end

local LootableComponent = {}
LootableComponent.__index = LootableComponent

function LootableComponent.new(scrap, pivot)
    local self = setmetatable({}, LootableComponent)
    self.object = scrap
    self.pivotPos = pivot
    self.bin = createBin()
    self.isAvailable = true
    local values = scrap:FindFirstChild("Values")
    if values then
        self.isAvailable = values:GetAttribute("Available") ~= false
    end
    self.container = Instance.new("Frame")
    self.nameLabel = Instance.new("TextLabel")
    self.container.Visible = false
    self.container.AnchorPoint = Vector2.new(0.5, 0)
    self.container.BackgroundTransparency = 1
    self.container.Size = UDim2.new(0, 100, 0, 50)
    self.nameLabel.BackgroundTransparency = 1
    self.nameLabel.Font = Enum.Font.Nunito
    self.nameLabel.Text = "Scrap"
    self.nameLabel.TextColor3 = Color3.new(0, 1, 0)
    self.nameLabel.TextSize = 15
    self.nameLabel.TextStrokeTransparency = 0.5
    self.nameLabel.Size = UDim2.new(1, 0, 0, 14)
    self.nameLabel.Parent = self.container
    if scrapScreenGui then self.container.Parent = scrapScreenGui end
    if values then
        local conn = values:GetAttributeChangedSignal("Available"):Connect(function()
            self.isAvailable = values:GetAttribute("Available")
        end)
        self.bin:add(conn)
    end
    local renderConn = game:GetService("RunService").RenderStepped:Connect(function() self:render() end)
    self.bin:add(renderConn)
    local deathConn = scrap.AncestryChanged:Connect(function()
        if not scrap or not scrap.Parent then self:destroy() end
    end)
    self.bin:add(deathConn)
    return self
end

function LootableComponent:render()
    local camera = workspace.CurrentCamera
    local player = game.Players.LocalPlayer
    local character = player.Character
    if not camera or not character then
        self.container.Visible = false
        return
    end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then
        self.container.Visible = false
        return
    end
    if not self.object or not self.object.Parent then
        self:destroy()
        return
    end
    local position, visible = camera:WorldToViewportPoint(self.pivotPos.Position)
    if visible and self.isAvailable then
        local scale = 1 / (position.Z * math.tan(math.rad(camera.FieldOfView * 0.5)) * 2) * 1000
        local width, height = math.floor(1 * scale), math.floor(3 * scale)
        local x, y = math.floor(position.X), math.floor(position.Y)
        local xPosition, yPosition = math.floor(x - width * 0.5), math.floor((y - height * 0.5) + (0.5 * scale))
        self.container.Visible = true
        self.container.Position = UDim2.new(0, xPosition, 0, yPosition)
        local distance = (hrp.Position - self.pivotPos.Position).Magnitude
        self.nameLabel.Text = string.format("Scrap [%.1f]", distance)
    else
        self.container.Visible = false
    end
end

function LootableComponent:destroy()
    if self.bin then pcall(function() self.bin:destroy() end) self.bin = nil end
    if self.container then pcall(function() self.container:Destroy() end) self.container = nil end
    self.object = nil
end

        local location = findLocation(locationData.path)
        if location then
            local attachPart = location:FindFirstChild("HumanoidRootPart") 
                or location:FindFirstChild("Head") 
                or location:FindFirstChildOfClass("BasePart")
            if not attachPart then attachPart = location end
            
            local billboard = Instance.new("BillboardGui")
            billboard.Name = "LocationESP_" .. locationData.name
            billboard.Size = UDim2.new(0, 200, 0, 40)
            billboard.StudsOffset = Vector3.new(0, 5, 0)
            billboard.AlwaysOnTop = true
            billboard.Parent = attachPart
            
            local textLabel = Instance.new("TextLabel")
            textLabel.Size = UDim2.new(1, 0, 1, 0)
            textLabel.BackgroundTransparency = 1
            textLabel.Text = locationData.name
            textLabel.TextColor3 = Color3.fromRGB(0, 255, 255)
            textLabel.TextStrokeTransparency = 0
            textLabel.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
            textLabel.TextScaled = true
            textLabel.Font = Enum.Font.GothamBold
            textLabel.Parent = billboard
            
            local conn
            conn = location.AncestryChanged:Connect(function()
                if not location or not location.Parent then
                    pcall(function() billboard:Destroy() end)
                    if conn then 
                        pcall(function() conn:Disconnect() end)
                    end
                end
            end)
            
            table.insert(locationESPObjects, {
                name = locationData.name,
                location = location,
                billboard = billboard,
                textLabel = textLabel,
                connection = conn
            })
            
            locationESPBin:add(conn)
        end
    end
end

local function toggleLocationESP(Value)
    locationESPEnabled = Value
    if locationESPScanConnection then
        pcall(function() locationESPScanConnection:Disconnect() end)
        locationESPScanConnection = nil
    end
    if Value then
        setupLocationESP()
        locationESPScanConnection = game:GetService("RunService").Heartbeat:Connect(function()
            if locationESPEnabled then
                local needsRescan = false
                for _, data in ipairs(locationESPObjects) do
                    if not data.location or not data.location.Parent then
                        needsRescan = true
                        break
                    end
                end
                if needsRescan then setupLocationESP() end
            else
                if locationESPScanConnection then
                    pcall(function() locationESPScanConnection:Disconnect() end)
                    locationESPScanConnection = nil
                end
            end
        end)
    else
        clearLocationESP()
    end
end

   Name = "Location ESP",
   CurrentValue = false,
   Flag = "LocationESP",
   Callback = function(Value) toggleLocationESP(Value) end,
})

-- ===== CHAIN ESP =====
local chainESPEnabled = false
local chainHighlight = nil
local chainConnection = nil
local chainCheckConnection = nil
local chainStatusBillboard = nil
local chainStatusLabels = {}

local MAX_ANGER = 190

local function calculateHealthFromAnger(anger)
    if anger <= 0 then
        return 100
    end
    local healthPercent = 100 - (anger / MAX_ANGER * 100)
    return math.max(0, math.min(100, healthPercent))
end

local function updateChainStatusDisplay(chain)
    if not chain then return end
    if chainStatusBillboard then
        pcall(function() chainStatusBillboard:Destroy() end)
        chainStatusBillboard = nil
    end
    chainStatusLabels = {}
    
    local hrp = chain:FindFirstChild("HumanoidRootPart") or chain:FindFirstChild("Head") or chain:FindFirstChild("Torso")
    if not hrp then return end
    
    local isClashing = chain:GetAttribute("Clashing") == true
    local isQTEImmune = chain:GetAttribute("QTEImmunity") == true
    local anger = chain:GetAttribute("Anger") or 0
    local health = calculateHealthFromAnger(anger)
    local healthText = string.format("Health: %.1f%%", health)
    
    local hasAnyStatus = isClashing or isQTEImmune
    
    local billboard = Instance.new("BillboardGui")
    billboard.Name = "ChainStatus"
    billboard.Size = UDim2.new(0, 200, 0, hasAnyStatus and 85 or 30)
    billboard.StudsOffset = Vector3.new(0, 3, 0)
    billboard.AlwaysOnTop = true
    billboard.Parent = hrp
    
    local layout = Instance.new("UIListLayout")
    layout.FillDirection = Enum.FillDirection.Vertical
    layout.HorizontalAlignment = Enum.HorizontalAlignment.Center
    layout.VerticalAlignment = Enum.VerticalAlignment.Top
    layout.Padding = UDim.new(0, 2)
    layout.Parent = billboard
    
    local healthLabel = Instance.new("TextLabel")
    healthLabel.Size = UDim2.new(1, 0, 0, 25)
    healthLabel.BackgroundTransparency = 1
    healthLabel.Text = healthText
    
    if health > 50 then
        healthLabel.TextColor3 = Color3.fromRGB(0, 255, 0)
    elseif health > 25 then
        healthLabel.TextColor3 = Color3.fromRGB(255, 165, 0)
    else
        healthLabel.TextColor3 = Color3.fromRGB(255, 0, 0)
    end
    
    healthLabel.TextStrokeTransparency = 0
    healthLabel.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    healthLabel.TextScaled = true
    healthLabel.Font = Enum.Font.GothamBold
    healthLabel.Parent = billboard
    table.insert(chainStatusLabels, healthLabel)
    
    if isClashing then
        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, 0, 0, 25)
        label.BackgroundTransparency = 1
        label.Text = "Clashing"
        label.TextColor3 = Color3.fromRGB(255, 100, 100)
        label.TextStrokeTransparency = 0
        label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
        label.TextScaled = true
        label.Font = Enum.Font.GothamBold
        label.Parent = billboard
        table.insert(chainStatusLabels, label)
    end
    
    if isQTEImmune then
        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, 0, 0, 25)
        label.BackgroundTransparency = 1
        label.Text = "QTE Immune"
        label.TextColor3 = Color3.fromRGB(255, 255, 0)
        label.TextStrokeTransparency = 0
        label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
        label.TextScaled = true
        label.Font = Enum.Font.GothamBold
        label.Parent = billboard
        table.insert(chainStatusLabels, label)
    end
    
    chainStatusBillboard = billboard
end

local function findChain()
    local chain = workspace:FindFirstChild("Misc")
    if chain then chain = chain:FindFirstChild("AI") end
    if chain then chain = chain:FindFirstChild("CHAIN") end
    return chain
end

local function updateChainESP()
    if chainHighlight then
        pcall(function() chainHighlight:Destroy() end)
        chainHighlight = nil
    end
    if chainConnection then
        pcall(function() chainConnection:Disconnect() end)
        chainConnection = nil
    end
    if chainStatusBillboard then
        pcall(function() chainStatusBillboard:Destroy() end)
        chainStatusBillboard = nil
    end
    chainStatusLabels = {}
    
    if not chainESPEnabled then return end
    
    local chain = findChain()
    if chain and chain:IsA("Model") then
        chainHighlight = Instance.new("Highlight")
        chainHighlight.Parent = chain
        chainHighlight.FillColor = Color3.fromRGB(255, 21, 21)
        chainHighlight.OutlineColor = Color3.fromRGB(255, 255, 255)
        chainHighlight.FillTransparency = 0.60
        chainHighlight.OutlineTransparency = 0
        
        updateChainStatusDisplay(chain)
        
        local function onAttributeChanged()
            if chain and chain.Parent then
                updateChainStatusDisplay(chain)
            end
        end
        
        local clashingConn = chain:GetAttributeChangedSignal("Clashing"):Connect(onAttributeChanged)
        local qteConn = chain:GetAttributeChangedSignal("QTEImmunity"):Connect(onAttributeChanged)
        local angerConn = chain:GetAttributeChangedSignal("Anger"):Connect(onAttributeChanged)
        
        chainStatusLabels = {clashingConn, qteConn, angerConn}
        
        chainConnection = chain.AncestryChanged:Connect(function()
            if not chain or not chain.Parent then
                if chainHighlight then
                    pcall(function() chainHighlight:Destroy() end)
                    chainHighlight = nil
                end
                if chainConnection then
                    pcall(function() chainConnection:Disconnect() end)
                    chainConnection = nil
                end
                if chainStatusBillboard then
                    pcall(function() chainStatusBillboard:Destroy() end)
                    chainStatusBillboard = nil
                end
                for _, conn in ipairs(chainStatusLabels) do
                    if conn then 
                        pcall(function() conn:Disconnect() end)
                    end
                end
                chainStatusLabels = {}
            end
        end)
    end
end

local Toggle = Tab:CreateToggle({

   Callback = function(Value)
      chainESPEnabled = Value

            chainHighlight = nil
         end
         if chainConnection then
            pcall(function() chainConnection:Disconnect() end)
            chainConnection = nil
         end
         if chainStatusBillboard then
            pcall(function() chainStatusBillboard:Destroy() end)
            chainStatusBillboard = nil
         end
         for _, conn in ipairs(chainStatusLabels) do
            if conn then 
               pcall(function() conn:Disconnect() end)
            end
         end
         chainStatusLabels = {}
      end
   end,
})
