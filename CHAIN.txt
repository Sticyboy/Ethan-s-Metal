-- Function to press a key
local function pressKey(key)
    print("Attempting to press key: " .. key)
    
    -- Handle special keys
    local keyMap = {
        ["Space"] = Enum.KeyCode.Space,
        ["Enter"] = Enum.KeyCode.Return,
        ["Shift"] = Enum.KeyCode.LeftShift,
        ["Control"] = Enum.KeyCode.LeftControl,
        ["Alt"] = Enum.KeyCode.LeftAlt,
        ["Tab"] = Enum.KeyCode.Tab,
        ["Backspace"] = Enum.KeyCode.Backspace,
        ["Delete"] = Enum.KeyCode.Delete,
        ["Escape"] = Enum.KeyCode.Escape,
        ["Up"] = Enum.KeyCode.Up,
        ["Down"] = Enum.KeyCode.Down,
        ["Left"] = Enum.KeyCode.Left,
        ["Right"] = Enum.KeyCode.Right,
    }
    
    local keyCode
    
    -- Check if it's a special key
    if keyMap[key] then
        keyCode = keyMap[key]
    else
        -- Try to get the key code directly
        keyCode = Enum.KeyCode[key]
        
        -- If that fails, try to find it by name
        if not keyCode then
            for _, enum in ipairs(Enum.KeyCode:GetEnumItems()) do
                if enum.Name == key then
                    keyCode = enum
                    break
                end
            end
        end
    end
    
    if keyCode then
        print("Pressing key: " .. tostring(keyCode))
        VirtualInputManager:SendKeyEvent(true, keyCode, false, game)
        task.wait(0.05)
        VirtualInputManager:SendKeyEvent(false, keyCode, false, game)
        print("Pressed key: " .. key)
    else
        print("Could not find key: " .. key)
    end
end

-- Function to monitor a specific QTE
local function monitorQTE(qtePath, qteName)
    local qteText = game.Players.LocalPlayer.PlayerGui:FindFirstChild("Ingame")
    if qteText then
        qteText = qteText:FindFirstChild("MechanicsFrame")
        if qteText then
            qteText = qteText:FindFirstChild(qtePath)
            if qteText then
                qteText = qteText:FindFirstChild("ImageButton")
                if qteText then
                    qteText = qteText:FindFirstChild("PC")
                end
            end
        end
    end
    
    if not qteText then
        warn("QTE UI not found: " .. qteName)
        return false
    end
    
    print("QTE monitor started for: " .. qteName)
    
    -- Property change listener
    local connection = qteText:GetPropertyChangedSignal("Text"):Connect(function()
        if not autoQTEEnabled then return end
        local currentText = qteText.Text
        if currentText ~= "Hi" and currentText ~= "" then
            print("[" .. qteName .. "] QTE Key detected: " .. currentText)
            pressKey(currentText)
            task.wait(0.1)
            qteText.Text = "Hi"
        end
    end)
    table.insert(qteConnections, connection)
    
    -- Polling loop as backup
    local thread = task.spawn(function()
        while autoQTEEnabled do
            task.wait(0.1)
            if not autoQTEEnabled then break end
            local currentText = qteText.Text
            if currentText ~= "Hi" and currentText ~= "" then
                print("[" .. qteName .. "] QTE Key detected (poll): " .. currentText)
                pressKey(currentText)
                task.wait(0.1)
                qteText.Text = "Hi"
            end
        end
    end)
    table.insert(qtePollingThreads, thread)
    
    return true
end

-- Toggle function

    autoQTEEnabled = Value
    
    if Value then

        for _, conn in ipairs(qteConnections) do

        end
        qteConnections = {}
        
        for _, thread in ipairs(qtePollingThreads) do
            task.cancel(thread)
        end
        qtePollingThreads = {}
        
        -- Start monitoring both QTEs

        end
        qteConnections = {}
        
        for _, thread in ipairs(qtePollingThreads) do
            task.cancel(thread)
        end
        qtePollingThreads = {}
    end
end

-- Function to check distance to CHAIN
local function isNearChain()

    return distance <= 15

    task.wait(0.05)
    VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.Q, false, game)

-- Process sound for dodge choke
local function hookSound(sound)
    if not sound:IsA("Sound") then return end
    
    local function checkAndTrigger()
        if not autoDodgeChokeEnabled then return end
        
        local soundId = tostring(sound.SoundId)
        if soundId:find(TARGET_AUDIO_ID) then
            local now = os.clock()
            if (now - dodgeChokeCooldown) >= DODGE_COOLDOWN then
                -- Check if player is near CHAIN (within 15 studs)
                if isNearChain() then
                    dodgeChokeCooldown = now
                    task.spawn(pressQKeyWithDelay) -- Dodge with 0.3s delay
                    print("Auto Dodge Choke triggered! (0.3s delay)")
                else
                    local chain = workspace:FindFirstChild("Misc")
                    if chain then
                        chain = chain:FindFirstChild("AI")
                        if chain then
                            chain = chain:FindFirstChild("CHAIN")
                        end
                    end
                    if chain and chain:FindFirstChild("HumanoidRootPart") then
                        local player = game.Players.LocalPlayer
                        local character = player.Character
                        if character and character:FindFirstChild("HumanoidRootPart") then
                            local distance = (character.HumanoidRootPart.Position - chain.HumanoidRootPart.Position).Magnitude
                            print("CHAIN too far (" .. math.floor(distance) .. " studs) - not dodging")
                        end
                    end
                end
            end
        end
    end
    
    -- Connect to Played event
    sound.Played:Connect(checkAndTrigger)
    
    -- Also check when IsPlaying changes
    sound:GetPropertyChangedSignal("IsPlaying"):Connect(function()
        if sound.IsPlaying then
            checkAndTrigger()
        end
    end)
    
    table.insert(soundConnections, sound)
end

-- Setup sound hooks

    -- Hook all existing sounds
    for _, desc in pairs(game:GetDescendants()) do

    -- Hook new sounds as they're added
    game.DescendantAdded:Connect(function(desc)
        pcall(function()
            hookSound(desc)
        end)
    end)
end

-- Utility function to extract numeric ID from animation
local function extractNumericAnimationId(animationId)
    if not animationId then return nil end
    local id = tostring(animationId)
    local num = id:match("rbxassetid://(%d+)") or id:match("://(%d+)") or id:match("^(%d+)$")
    return num
end

    local function getAllAnimators(model)
        local animators = {}
        for _, descendant in ipairs(model:GetDescendants()) do

                table.insert(animators, descendant)
            end
        end
        return animators
    end
    
    local allAnimators = getAllAnimators(chain)
    
    if #allAnimators == 0 then

            end

                    end
                    break
                end
            end
        end
        
        -- Listen for new animations
        animator.AnimationPlayed:Connect(function(animationTrack)
            if not autoBladeClashEnabled then return end
            
            local animId = extractNumericAnimationId(animationTrack.Animation.AnimationId)
            for _, clashAnim in ipairs(BLADE_CLASH_ANIMATIONS) do
                if animId == clashAnim then
                    if isChainInRange() then
                        local now = os.clock()
                        if (now - bladeClashCooldown) >= BLADE_CLASH_COOLDOWN then
                            bladeClashCooldown = now
                            leftClick()

                                if chain then

                                end
                            end

-- ===== SCRAP ESP (HIGHLIGHT VERSION) =====
local scrapESPEnabled = false
local scrapHighlightConnections = {}

local function clearScrapHighlights()
    for _, data in pairs(scrapHighlightConnections) do
        if data.highlight then
            pcall(function() data.highlight:Destroy() end)
        end
        if data.connection then
            pcall(function() data.connection:Disconnect() end)
        end
    end
    scrapHighlightConnections = {}
end

local function updateScrapESP()
    clearScrapHighlights()
    
    if not scrapESPEnabled then return end
    
    local scrapFolder = workspace:FindFirstChild("Misc")
    if scrapFolder then
        scrapFolder = scrapFolder:FindFirstChild("Zones")
        if scrapFolder then
            scrapFolder = scrapFolder:FindFirstChild("LootingItems")
            if scrapFolder then
                scrapFolder = scrapFolder:FindFirstChild("Scrap")
            end
        end
    end
    
    if not scrapFolder then return end
    
    for _, scrapItem in ipairs(scrapFolder:GetChildren()) do
        if scrapItem:IsA("Model") then
            local highlight = Instance.new("Highlight")
            highlight.Parent = scrapItem
            highlight.FillColor = Color3.fromRGB(255, 255, 0)
            highlight.OutlineColor = Color3.fromRGB(255, 200, 0)
            highlight.FillTransparency = 0.5
            highlight.OutlineTransparency = 0.3
            
            local connection
            connection = scrapItem.AncestryChanged:Connect(function()
                if not scrapItem or not scrapItem.Parent then
                    if connection then connection:Disconnect() end
                    pcall(function() highlight:Destroy() end)
                end
            end)
            
            scrapHighlightConnections[scrapItem] = {
                highlight = highlight,
                connection = connection
            }
        end
    end
end

-- ===== CHAIN ESP (FIXED - properly re-adds highlight when CHAIN reappears) =====
local chainESPEnabled = false
local chainHighlight = nil
local chainConnection = nil
local chainCheckConnection = nil

local function updateChainESP()
    -- Remove existing highlight
    if chainHighlight then
        pcall(function() chainHighlight:Destroy() end)
        chainHighlight = nil
    end
    
    if chainConnection then
        pcall(function() chainConnection:Disconnect() end)
        chainConnection = nil
    end
    
    if not chainESPEnabled then return end
    
    -- Find CHAIN
    local chain = workspace:FindFirstChild("Misc")
    if chain then
        chain = chain:FindFirstChild("AI")
        if chain then
            chain = chain:FindFirstChild("CHAIN")
        end
    end
    
    if chain and chain:IsA("Model") then
        -- Create highlight
        chainHighlight = Instance.new("Highlight")
        chainHighlight.Parent = chain
        chainHighlight.FillColor = Color3.fromRGB(255, 21, 21)
        chainHighlight.OutlineColor = Color3.fromRGB(255, 255, 255)
        chainHighlight.FillTransparency = 0.60
        chainHighlight.OutlineTransparency = 0
        
        -- Watch for CHAIN being removed
        chainConnection = chain.AncestryChanged:Connect(function()
            if not chain or not chain.Parent then
                if chainHighlight then
                    pcall(function() chainHighlight:Destroy() end)
                    chainHighlight = nil
                end
                if chainConnection then
                    pcall(function() chainConnection:Disconnect() end)
                    chainConnection = nil
                end
            end
        end)
    end
end

local Toggle = Tab:CreateToggle({
   Name = "Chain ESP - red",
   CurrentValue = false,
   Flag = "ChainESP",
   Callback = function(Value)
      chainESPEnabled = Value
      
      -- Stop the check loop
      if chainCheckConnection then
         pcall(function() chainCheckConnection:Disconnect() end)
         chainCheckConnection = nil
      end
      
      if Value then
         -- Initial setup
         updateChainESP()
         
         -- Keep checking for CHAIN to reappear
         chainCheckConnection = game:GetService("RunService").Heartbeat:Connect(function()
            if chainESPEnabled then
               -- If highlight doesn't exist but CHAIN does, re-add it
               if not chainHighlight then
                  local chain = workspace:FindFirstChild("Misc")
                  if chain then
                     chain = chain:FindFirstChild("AI")
                     if chain then
                        chain = chain:FindFirstChild("CHAIN")
                     end
                  end
                  
                  if chain and chain:IsA("Model") then
                     updateChainESP()
                  end
               end
            else
               if chainCheckConnection then
                  pcall(function() chainCheckConnection:Disconnect() end)
                  chainCheckConnection = nil
               end
            end
         end)
      else
         -- Clean up
         if chainHighlight then
            pcall(function() chainHighlight:Destroy() end)
            chainHighlight = nil
         end
         if chainConnection then
            pcall(function() chainConnection:Disconnect() end)
            chainConnection = nil
         end
      end
   end,
})
