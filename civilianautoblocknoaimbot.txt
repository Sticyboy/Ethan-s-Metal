local function getTeamFolder(name)
    local gameAssets = Workspace:FindFirstChild("GameAssets")
    if not gameAssets then return nil end
    local teams = gameAssets:FindFirstChild("Teams")
    if not teams then return nil end
    return teams:FindFirstChild(name)

    if not Character then return nil end

        if c.Name == lp.Name then return true end

local function getAbilityBoundToKey(keyCode)
    local folder = getAbilitiesFolder()
    if not folder then return nil, nil end
    local target = keyCode.Name
    for _, ability in ipairs(folder:GetChildren()) do
        local input = ability:FindFirstChild("Input")
        if input then
            local txt = (input.Text or ""):gsub("%s", "")
            if txt == target then
                return ability.Name, ability
            end
        end
    end
    return nil, nil
end

-- ===== MODE DETECTION =====
local function getPursuerMode()
    local folder = getAbilitiesFolder()
    if not folder then return nil end
    local cleave = folder:FindFirstChild("Cleave")
    local t1 = cleave and cleave:FindFirstChild("Title")
    if t1 then
        if t1.Text == "Cleave" then return "Normal" end
        if t1.Text == "Lunge" then return "Stalk" end
    end
    local howl = folder:FindFirstChild("Howl")
    local t2 = howl and howl:FindFirstChild("Title")
    if t2 then
        if t2.Text == "Howl" then return "Normal" end
        if t2.Text == "Chomp" then return "Stalk" end
    end
    return nil
end

local function getHarkenMode()
    local folder = getAbilitiesFolder()

    local repress = folder:FindFirstChild("Repress")

-- ===== SHIFTLOCK =====
local shiftlockWasEnabled = false
local shiftlockActiveFromAimbot = false

local function isShiftLockEnabled()
    return UserInputService.MouseBehavior == Enum.MouseBehavior.LockCenter
end

local function setShiftLock(enabled, fromAimbot)
    if enabled then
        if fromAimbot then
            shiftlockWasEnabled = isShiftLockEnabled()
            shiftlockActiveFromAimbot = true
        end
        if not isShiftLockEnabled() then
            pcall(function()
                local keybinds = lp:FindFirstChild("PlayerData")
                    and lp.PlayerData:FindFirstChild("Settings")
                    and lp.PlayerData.Settings:FindFirstChild("Keybinds")
                local sk = keybinds and keybinds:FindFirstChild("ShiftLock")
                if sk then
                    local kc = Enum.KeyCode[sk.Value]
                    if kc then
                        VirtualInputManager:SendKeyEvent(true, kc, false, game)
                        VirtualInputManager:SendKeyEvent(false, kc, false, game)
                        return
                    end
                end
                VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.LeftControl, false, game)
                VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.LeftControl, false, game)
            end)
        end
    else
        if fromAimbot and shiftlockActiveFromAimbot then
            if not shiftlockWasEnabled then
                if isShiftLockEnabled() then
                    pcall(function()
                        local keybinds = lp:FindFirstChild("PlayerData")
                            and lp.PlayerData:FindFirstChild("Settings")
                            and lp.PlayerData.Settings:FindFirstChild("Keybinds")
                        local sk = keybinds and keybinds:FindFirstChild("ShiftLock")
                        if sk then
                            local kc = Enum.KeyCode[sk.Value]
                            if kc then
                                VirtualInputManager:SendKeyEvent(true, kc, false, game)
                                VirtualInputManager:SendKeyEvent(false, kc, false, game)
                                return
                            end
                        end
                        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.LeftControl, false, game)
                        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.LeftControl, false, game)
                    end)
                end
            end
            shiftlockActiveFromAimbot = false
        end
    end
end

        return

    clearIndicatorLabels()
    addIndicatorLabel(
        "Autoblock: " .. (autoBlockEnabled and "True" or "False"),
        autoBlockEnabled and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(255, 60, 60),
        1
    )
end

local function isFacing(localRoot, targetRoot)
    if not facingCheckEnabled then return true end
    local dx = localRoot.Position.X - targetRoot.Position.X
    local dy = localRoot.Position.Y - targetRoot.Position.Y
    local dz = localRoot.Position.Z - targetRoot.Position.Z
    local mag = math.sqrt(dx*dx + dy*dy + dz*dz)
    if mag == 0 then return true end
    local inv = 1 / mag
    local ux, uy, uz = dx * inv, dy * inv, dz * inv
    local lv = targetRoot.CFrame.LookVector
    return (lv.X * ux + lv.Y * uy + lv.Z * uz) > customFacingDot
end

local function fireBlock()
    if not amISurvivor() then return end
    if not hasBlockAbility() then return end
    pcall(function() UseAbility:InvokeServer("Block") end)
end

local function safeDestroyCircle(c)
    if not c then return end
    pcall(function() if c.Parent then c:Destroy() end end)
end

local function updateRangeVisuals()
    local now = tick()
    if now - lastVisualUpdate < VISUAL_UPDATE_COOLDOWN then return end
    lastVisualUpdate = now

    for _, c in pairs(detectionCircles) do safeDestroyCircle(c) end
    detectionCircles = {}

    if not rangeVisualEnabled then return end

    local myChar = lp.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end

    local killerFolder = getKillerFolder()
    if not killerFolder then return end

    for _, killer in ipairs(killerFolder:GetChildren()) do
        local killerName = killer:GetAttribute("KillerName")
        if killerName and TRACKED_SOUNDS[killerName] then
            local hrp = killer:FindFirstChild("HumanoidRootPart")
            if hrp and not hrp:FindFirstChild("AutoblockRangeCircle") then
                local inRange = (hrp.Position - myRoot.Position).Magnitude <= detectionRange
                pcall(function()
                    local c = Instance.new("CylinderHandleAdornment")
                    c.Name = "AutoblockRangeCircle"
                    c.Adornee = hrp
                    c.Color3 = inRange and Color3.fromRGB(255, 0, 0) or Color3.fromRGB(255, 165, 0)
                    c.AlwaysOnTop = true
                    c.ZIndex = 1
                    c.Transparency = 0.6
                    c.Radius = detectionRange
                    c.Height = 0.12
                    c.CFrame = CFrame.new(0, -(hrp.Size.Y / 2 + 0.05), 0) * CFrame.Angles(math.rad(90), 0, 0)
                    c.Parent = hrp
                    detectionCircles[killer] = c
                end)
            end
        end
    end
end

local function isKillerAlive(killer)
    local humanoid = killer:FindFirstChildOfClass("Humanoid")
    if not humanoid then return false end
    local health = humanoid:GetAttribute("Health") or humanoid.Health
    return health and health > 0
end

local function isKillerInRange(killer)
    local myChar = lp.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return false end
    local hrp = killer:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    return (hrp.Position - myRoot.Position).Magnitude <= detectionRange
end

local function allGatesPass(killer)
    if not autoBlockEnabled then return false end
    if not hasBlockAbility() then return false end
    if not amISurvivor() then return false end
    if not isKillerAlive(killer) then return false end
    if not isKillerInRange(killer) then return false end
    local myRoot = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    local killerRoot = killer:FindFirstChild("HumanoidRootPart")
    if not myRoot or not killerRoot then return false end
    if not isFacing(myRoot, killerRoot) then return false end
    return true
end

local function hookSound(killer, sound)
    if not sound:IsA("Sound") then return end
    if sound:GetAttribute("AutoblockHooked") then return end
    sound:SetAttribute("AutoblockHooked", true)

    sound.Played:Connect(function()
        if not allGatesPass(killer) then return end
        local killerName = killer:GetAttribute("KillerName")
        local waitSeconds = nil
        if killerName and SOUNDS_THAT_WAIT[killerName] then
            waitSeconds = SOUNDS_THAT_WAIT[killerName][sound.Name]
        end
        if type(waitSeconds) ~= "number" or waitSeconds <= 0 then
            fireBlock()
            return
        end
        task.spawn(function()
            local startTime = tick()
            while true do
                if tick() - startTime > 10 then return end
                if not sound.Playing and sound.TimePosition < waitSeconds then return end
                if sound.TimePosition >= waitSeconds then
                    if allGatesPass(killer) then fireBlock() end
                    return
                end
                RunService.Heartbeat:Wait()
            end
        end)
    end)
end

local function hookKillerSounds(killer)
    local hrp = killer:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local killerName = killer:GetAttribute("KillerName")
    local soundList = killerName and TRACKED_SOUNDS[killerName]
    if not soundList then return end
    for _, child in ipairs(hrp:GetChildren()) do
        if child:IsA("Sound") and soundList[child.Name] then
            hookSound(killer, child)
        end
    end
end

local function clearKillerData(killer)
    local data = killerData[killer]
    if data and data.soundConnections then
        for _, conn in ipairs(data.soundConnections) do
            pcall(function() conn:Disconnect() end)
        end
    end
    killerData[killer] = nil
    if detectionCircles[killer] then
        safeDestroyCircle(detectionCircles[killer])
        detectionCircles[killer] = nil
    end
    local hrp = killer and killer:FindFirstChild("HumanoidRootPart")
    if hrp then
        local c = hrp:FindFirstChild("AutoblockRangeCircle")
        if c then pcall(function() c:Destroy() end) end
    end
end

local function scanKillers()
    local killerFolder = getKillerFolder()
    if not killerFolder then return end
    local seen = {}
    for _, killer in ipairs(killerFolder:GetChildren()) do
        local killerName = killer:GetAttribute("KillerName")
        if killerName and TRACKED_SOUNDS[killerName] then
            seen[killer] = true
            if not killerData[killer] then
                killerData[killer] = { soundConnections = {}, killerName = killerName }
            end
            hookKillerSounds(killer)
            if not killer:GetAttribute("AutoblockScanBound") then
                killer:SetAttribute("AutoblockScanBound", true)
                task.spawn(function()
                    local hrp = killer:WaitForChild("HumanoidRootPart", 10)
                    if hrp then
                        hookKillerSounds(killer)
                        hrp.ChildAdded:Connect(function(child)
                            if child:IsA("Sound") then
                                local kName = killer:GetAttribute("KillerName")
                                local sList = kName and TRACKED_SOUNDS[kName]
                                if sList and sList[child.Name] then
                                    hookSound(killer, child)
                                end
                            end
                        end)
                    end
                end)
            end
        end
    end
    for killer, _ in pairs(killerData) do
        if not seen[killer] or not killer.Parent then
            clearKillerData(killer)
        end
    end
end

local function bindKillerFolder()
    local killerFolder = getKillerFolder()
    if not killerFolder then return end
    if killerFolder:GetAttribute("AutoblockFolderBound") then return end
    killerFolder:SetAttribute("AutoblockFolderBound", true)
    killerFolder.ChildAdded:Connect(function() task.wait(0.3); scanKillers() end)
    killerFolder.ChildRemoved:Connect(function(child) clearKillerData(child) end)
end

local function startAutoblock()
    if scanConnection then return end
    bindKillerFolder()
    scanKillers()
    scanConnection = RunService.Heartbeat:Connect(function()
        pcall(scanKillers)
        pcall(updateRangeVisuals)
        pcall(refreshIndicators)
    end)
end

local function stopAutoblock()
    if scanConnection then scanConnection:Disconnect(); scanConnection = nil end
    for _, c in pairs(detectionCircles) do safeDestroyCircle(c) end
    detectionCircles = {}
end

startAutoblock()

-- ===== PREDICTION =====
local function updatePositionHistory(character, pos)
    if not character or not pos then return end
    if not PositionHistory[character] then PositionHistory[character] = {} end
    local h = PositionHistory[character]
    table.insert(h, pos)
    if #h > HISTORY_SIZE then table.remove(h, 1) end
end

local function getMovementDirection(character)
    local h = PositionHistory[character]
    if not h or #h < 2 then return Vector3.new(0, 0, 0) end
    local overall = h[#h] - h[1]
    local recent = (#h >= 3) and (h[#h] - h[#h - 1]) or Vector3.new(0, 0, 0)
    if recent.Magnitude > MOVE_THRESHOLD then return recent.Unit end
    if overall.Magnitude > MOVE_THRESHOLD then return overall.Unit end
    return Vector3.new(0, 0, 0)
end

local function getPredictedPosition(character, multiplier)
    if not character then return nil end
    local root = character:FindFirstChild("HumanoidRootPart")
        or character:FindFirstChild("Head")
        or character:FindFirstChild("Torso")
    if not root then return nil end
    local pos = root.Position
    updatePositionHistory(character, pos)
    local dir = getMovementDirection(character)
    if dir.Magnitude > MOVE_THRESHOLD then
        return pos + dir * (BASE_PREDICTION * (multiplier or AimbotPredictionMul))
    end
    return pos
end

local function getValidTargets(targetTeam)
    local targets = {}
    if not Character or not HumanoidRootPart then return targets end
    local folder = getTeamFolder(targetTeam)
    if not folder then return targets end
    local viewport = Camera.ViewportSize
    local center = Vector2.new(viewport.X / 2, viewport.Y / 2)
    local myPos = HumanoidRootPart.Position
    for _, obj in ipairs(folder:GetChildren()) do
        if obj.Name ~= lp.Name then
            local hum = obj:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local part = obj:FindFirstChild("HumanoidRootPart")
                    or obj:FindFirstChild("Head") or obj:FindFirstChild("Torso")
                if part then
                    local dist = (part.Position - myPos).Magnitude
                    if dist <= AimbotMaxDistance then
                        local screenDist = 0
                        local onScreen = true
                        if AimbotUseFOV then
                            local sp, vis = Camera:WorldToViewportPoint(part.Position)
                            if vis then
                                screenDist = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                                if screenDist > AimbotFOV then onScreen = false end
                            else
                                onScreen = false
                            end
                        end
                        if onScreen then
                            table.insert(targets, { obj = obj, part = part, distance = dist, screenDistance = screenDist, health = hum.Health })
                        end
                    end
                end
            end
        end
    end
    if AimbotUseFOV then
        table.sort(targets, function(a, b) return a.screenDistance < b.screenDistance end)
    else
        table.sort(targets, function(a, b) return a.distance < b.distance end)
    end
    return targets
end

local function createTargetSphere(targetCharacter)
    if TargetSphere then pcall(function() TargetSphere:Destroy() end); TargetSphere = nil end
    if TargetSphereConn then pcall(function() TargetSphereConn:Disconnect() end); TargetSphereConn = nil end
    if not targetCharacter then return end
    local root = targetCharacter:FindFirstChild("HumanoidRootPart") or targetCharacter:FindFirstChild("Head") or targetCharacter:FindFirstChild("Torso")
    if not root then return end

    local sphere = Instance.new("Part")
    sphere.Shape = Enum.PartType.Ball
    sphere.Size = Vector3.new(2, 2, 2)
    sphere.Material = Enum.Material.Neon
    sphere.Color = Color3.fromRGB(255, 40, 40)
    sphere.Anchored = true
    sphere.CanCollide = false
    sphere.CanQuery = false
    sphere.CanTouch = false
    sphere.Name = "AimbotTargetSphere"
    sphere.Parent = Workspace
    sphere.Position = root.Position
    TargetSphere = sphere

    TargetSphereConn = RunService.Heartbeat:Connect(function()
        if not sphere.Parent or not targetCharacter or not targetCharacter.Parent then
            sphere:Destroy()
            if TargetSphereConn then TargetSphereConn:Disconnect() end
            TargetSphere = nil; TargetSphereConn = nil
            return
        end
        local hum = targetCharacter:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then
            sphere:Destroy()
            if TargetSphereConn then TargetSphereConn:Disconnect() end
            TargetSphere = nil; TargetSphereConn = nil
            return
        end
        local newRoot = targetCharacter:FindFirstChild("HumanoidRootPart") or targetCharacter:FindFirstChild("Head") or targetCharacter:FindFirstChild("Torso")
        if not newRoot then return end
        local pred = getPredictedPosition(targetCharacter, AimbotPredictionMul) or newRoot.Position
        sphere.Position = sphere.Position:Lerp(pred, 0.25)
    end)
end

local function cleanupTargetSphere()
    if TargetSphere then pcall(function() TargetSphere:Destroy() end); TargetSphere = nil end
    if TargetSphereConn then pcall(function() TargetSphereConn:Disconnect() end); TargetSphereConn = nil end
end

local function stopAimbot()
    AimbotActive = false
    AimbotTarget = nil
    PieChargeActive = false
    if AimbotActiveTimer then pcall(function() task.cancel(AimbotActiveTimer) end); AimbotActiveTimer = nil end
    if AimbotConnection then pcall(function() AimbotConnection:Disconnect() end); AimbotConnection = nil end
    cleanupTargetSphere()
    setShiftLock(false, true)
end

local function createFOVCircle()
    if AimbotFOVCircle then pcall(function() AimbotFOVCircle:Destroy() end); AimbotFOVCircle = nil end
    if not AimbotEnabled or not AimbotUseFOV then return end
    if not Drawing or not Drawing.new then return end
    local circle = Drawing.new("Circle")
    circle.Transparency = 1
    circle.Thickness = 2
    circle.NumSides = 60
    circle.Radius = AimbotFOV
    circle.Color = Color3.fromRGB(255, 255, 255)
    circle.Filled = false
    circle.Visible = true
    AimbotFOVCircle = circle
    RunService.RenderStepped:Connect(function()
        if AimbotFOVCircle and AimbotEnabled and AimbotUseFOV then
            local vp = Camera.ViewportSize
            AimbotFOVCircle.Position = Vector2.new(vp.X / 2, vp.Y / 2)
            AimbotFOVCircle.Radius = AimbotFOV
            AimbotFOVCircle.Visible = true
        elseif AimbotFOVCircle then
            AimbotFOVCircle.Visible = false
        end
    end)
end

local function startLockOnTarget(target, duration)
    if not target then return end
    if AimbotActive then return end
    AimbotActive = true
    AimbotTarget = target
    setShiftLock(true, true)
    createTargetSphere(target)
    local conn
    conn = RunService.RenderStepped:Connect(function()
        if not AimbotActive or not AimbotTarget then
            if conn then conn:Disconnect() end
            return
        end
        local hum = AimbotTarget:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then if conn then conn:Disconnect() end; return end
        local pred = getPredictedPosition(AimbotTarget, AimbotPredictionMul)
        if not pred then if conn then conn:Disconnect() end; return end
        local camPos = Camera.CFrame.Position
        local targetLook = CFrame.lookAt(camPos, pred).LookVector
        local smoothLook = Camera.CFrame.LookVector:Lerp(targetLook, AimbotSmoothness)
        Camera.CFrame = CFrame.new(camPos, camPos + smoothLook)
    end)
    AimbotConnection = conn
    if AimbotActiveTimer then pcall(function() task.cancel(AimbotActiveTimer) end) end
    if duration then
        AimbotActiveTimer = task.delay(duration, function() stopAimbot() end)
    else
        AimbotActiveTimer = nil
    end
end

local function runAimbotFor(duration, targetTeam)
    if not AimbotEnabled then return end
    if AimbotActive then return end
    if not Character or not HumanoidRootPart then return end
    targetTeam = targetTeam or "Survivor"
    local targets = getValidTargets(targetTeam)
    if #targets == 0 then return end
    startLockOnTarget(targets[1].obj, duration)
end

local function runSurvivorAimbot(duration, abilityObj, def)
    if not AimbotEnabled then return end
    if AimbotActive then return end
    if def and def.skipIfTitle then
        local title = abilityObj and abilityObj:FindFirstChild("Title")
        if title and title.Text == def.skipIfTitle then return end
    end
    local targetTeam = (def and def.target) or "Killer"
    runAimbotFor(duration, targetTeam)
end

local function startPieCharge()
    if AimbotActive then return end
    PieChargeActive = true
    local targets = getValidTargets("Killer")
    if #targets == 0 then PieChargeActive = false; return end
    startLockOnTarget(targets[1].obj, nil)
end

local function endPieCharge()
    if not PieChargeActive then return end
    PieChargeActive = false
    if AimbotActive then
        if AimbotActiveTimer then pcall(function() task.cancel(AimbotActiveTimer) end) end
        AimbotActiveTimer = task.delay(0.2, function() stopAimbot() end)
    end
end

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if not AimbotEnabled then return end
    if not Character then return end

    local myRole = getLocalRole()
    if input.KeyCode == Enum.KeyCode.Space then return end

    if myRole == "Killer" then
        local kName = getMyKillerName()
        if not kName then return end
        local binds = KILLER_AIMBOT_KEYS[kName]
        local duration = nil
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            duration = getM1Duration()
        elseif input.KeyCode == Enum.KeyCode.Q then
            if binds and binds.Q then duration = binds.Q() end
        elseif input.KeyCode == Enum.KeyCode.E then
            if binds and binds.E then duration = binds.E() end
        elseif input.KeyCode == Enum.KeyCode.R then
            if binds and binds.R then duration = binds.R() end
        elseif input.KeyCode == Enum.KeyCode.F then
            if binds and binds.F then duration = binds.F() end
        end
        if duration and duration > 0 then
            runAimbotFor(duration, "Survivor")
        end
        return
    end

    if myRole == "Survivor" then
        if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
        local abilityName, abilityObj = getAbilityBoundToKey(input.KeyCode)
        if not abilityName then return end
        local def = CIVILIAN_ABILITIES[abilityName]
        if not def then return end

        if def.special == "charge" then
            PieChargeActive = false
            _G.__AimbotPiePending = true
            task.delay(3, function() _G.__AimbotPiePending = false end)
            return
        end

        if abilityName == "Revolver" and hasLoveshotSynergy() then
            local title = abilityObj:FindFirstChild("Title")
            if not (title and title.Text == "Reload") then
                runSurvivorAimbot(2.5, abilityObj, def)
            end
            return
        end

        if def.duration and def.duration > 0 then
            runSurvivorAimbot(def.duration, abilityObj, def)
        end
        return
    end
end)

--  VISUALS - ESP (Highlight-based)

local ESPEnabled      = false
local ESPTransparency = 0.5
local ESPShowName     = true
local ESPShowHealth   = true
local ESPShowStamina  = true

local ESPHighlights = {}   -- [model] = { highlight = Highlight, billboard = BillboardGui, conn = RBXScriptConnection }

local function getESPColorForModel(model)
    local parent = model.Parent
    if not parent then return Color3.fromRGB(200, 200, 200) end
    local myRole = getLocalRole()
    local parentName = parent.Name  -- "Killer" / "Survivor" / "Ghost"

    if not myRole then return Color3.fromRGB(200, 200, 200) end

    -- Teammate = same team
    if (myRole == "Killer" and parentName == "Killer")
        or (myRole == "Survivor" and parentName == "Survivor") then
        return Color3.fromRGB(60, 255, 60)   -- green
    end

    -- Enemy
    if (myRole == "Killer" and parentName == "Survivor")
        or (myRole == "Survivor" and parentName == "Killer") then
        return Color3.fromRGB(255, 60, 60)   -- red
    end

    -- Ghost / other
    return Color3.fromRGB(220, 220, 220)
end

local function destroyESPFor(model)
    local data = ESPHighlights[model]
    if not data then return end
    if data.conn then pcall(function() data.conn:Disconnect() end) end
    if data.highlight then pcall(function() data.highlight:Destroy() end) end
    if data.billboard then pcall(function() data.billboard:Destroy() end) end
    ESPHighlights[model] = nil
end

local function buildESPFor(model)
    if not model then return end
    if model.Name == lp.Name then return end
    destroyESPFor(model)

    local hum = model:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return end

    local color = getESPColorForModel(model)

    -- Highlight
    local hl = Instance.new("Highlight")
    hl.Name = "ESP_Highlight"
    hl.Adornee = model
    hl.FillColor = color
    hl.OutlineColor = color
    hl.FillTransparency = ESPTransparency
    hl.OutlineTransparency = 0
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Parent = model

    -- Billboard with name / HP / SP
    local billboard, label
    if ESPShowName or ESPShowHealth or ESPShowStamina then
        local head = model:FindFirstChild("Head")
        if head then
            billboard = Instance.new("BillboardGui")
            billboard.Name = "ESP_Billboard"
            billboard.Adornee = head
            billboard.AlwaysOnTop = true
            billboard.Size = UDim2.new(0, 200, 0, 60)
            billboard.StudsOffset = Vector3.new(0, 2, 0)
            billboard.Parent = model

            label = Instance.new("TextLabel")
            label.Size = UDim2.new(1, 0, 1, 0)
            label.BackgroundTransparency = 1
            label.Font = Enum.Font.GothamBold
            label.TextScaled = true
            label.TextColor3 = color
            label.TextStrokeTransparency = 0
            label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
            label.Text = model.Name
            label.Parent = billboard
        end
    end

    -- Live updater
    local conn = RunService.Heartbeat:Connect(function()
        if not model or not model.Parent then
            destroyESPFor(model)
            return
        end
        -- refresh highlight color in case team changed
        if hl and hl.Parent then
            local newColor = getESPColorForModel(model)
            hl.FillColor = newColor
            hl.OutlineColor = newColor
            hl.FillTransparency = ESPTransparency
        end
        -- update billboard
        if label and label.Parent then
            local lines = {}
            if ESPShowName then table.insert(lines, model.Name) end
            if ESPShowHealth and hum then
                table.insert(lines, "HP: " .. math.floor(hum.Health))
            end
            if ESPShowStamina then
                local s = model:GetAttribute("Stamina")
                local ms = model:GetAttribute("MaxStamina")
                if s and ms then
                    table.insert(lines, "SP: " .. math.floor(s) .. "/" .. math.floor(ms))
                end
            end
            label.Text = table.concat(lines, "\n")
            local newColor = getESPColorForModel(model)
            label.TextColor3 = newColor
        end
    end)

    ESPHighlights[model] = { highlight = hl, billboard = billboard, conn = conn }
end

local function scanESP()
    if not ESPEnabled then return end

    -- Build for known team folders
    local teams = {
        getKillerFolder(),
        getSurvivorFolder(),
        getTeamFolder("Ghost"),
    }

    local seen = {}
    for _, folder in ipairs(teams) do
        if folder then
            for _, model in ipairs(folder:GetChildren()) do
                if model:IsA("Model") then
                    seen[model] = true
                    if not ESPHighlights[model] then
                        local hum = model:FindFirstChildOfClass("Humanoid")
                        if hum and hum.Health > 0 then
                            buildESPFor(model)
                        end
                    end
                end
            end
        end
    end

    -- Clean up destroyed
    for model, _ in pairs(ESPHighlights) do
        if not seen[model] or not model.Parent then
            destroyESPFor(model)
        end
    end
end

local function refreshAllESP()
    for model, _ in pairs(ESPHighlights) do
        destroyESPFor(model)
    end
    if ESPEnabled then scanESP() end
end

local espScanConnection = nil
local function startESP()
    if espScanConnection then return end
    espScanConnection = RunService.Heartbeat:Connect(function()
        pcall(scanESP)
    end)
end
startESP()

-- Player add/remove re-scan
Players.PlayerAdded:Connect(function() task.wait(0.5) if ESPEnabled then scanESP() end end)
Players.PlayerRemoving:Connect(function(plr)
    for model, _ in pairs(ESPHighlights) do
        if model.Name == plr.Name then
            destroyESPFor(model)
        end
    end
end)
