local function shouldBlock()
    if blockChance >= 100 then return true end
    if blockChance <= 0 then return false end
    return math.random(1, 100) <= blockChance
end

local function isFacing(localRoot, targetRoot)
    if not facingCheckEnabled then return true end
    local dx = localRoot.Position.X - targetRoot.Position.X
    local dy = localRoot.Position.Y - targetRoot.Position.Y
    local dz = localRoot.Position.Z - targetRoot.Position.Z
    local mag = math.sqrt(dx*dx + dy*dy + dz*dz)
    if mag == 0 then return true end
    local invMag = 1 / mag
    local ux, uy, uz = dx * invMag, dy * invMag, dz * invMag
    local lv = targetRoot.CFrame.LookVector
    return (lv.X * ux + lv.Y * uy + lv.Z * uz) > customFacingDot
end

local function extractNumericSoundId(sound)
    if not sound then return nil end
    local sid = tostring(sound.SoundId)
    return sid:match("rbxassetid://(%d+)") or sid:match("://(%d+)") or sid:match("^(%d+)$")
end

local function extractNumericAnimationId(animationId)
    if not animationId then return nil end
    local id = tostring(animationId)
    return id:match("rbxassetid://(%d+)") or id:match("://(%d+)") or id:match("^(%d+)$")
end

local function getMyCharacterInfo()
    if SurvivorsFolder then
        for _, s in ipairs(SurvivorsFolder:GetChildren()) do
            local u = s:GetAttribute("Username")
            if u and u == lp.Name then return "Survivor", s.Name, s end
        end
    end
    if KillersFolder then
        for _, k in ipairs(KillersFolder:GetChildren()) do
            local u = k:GetAttribute("Username")
            if u and u == lp.Name then return "Killer", k.Name, k end
        end
    end
    return nil, nil, nil
end

    if name == "Guest1337" or name == "Shedletsky" or name == "TwoTime" or name == "Chance" or name == "JaneDoe" or name == "Elliot" or name == "007n7" or name == "7n7" then
        if SurvivorsFolder then
            for _, s in ipairs(SurvivorsFolder:GetChildren()) do
                if s.Name == name then table.insert(out, s) end
            end
        end
    else
        if KillersFolder then
            for _, k in ipairs(KillersFolder:GetChildren()) do
                if k.Name == name then table.insert(out, k) end
            end
        end
    end
    return out
end

local function updatePositionHistory(character, position)
    if not character or not position then return end
    if not positionHistory[character] then positionHistory[character] = {} end
    local h = positionHistory[character]
    table.insert(h, position)
    if #h > HISTORY_SIZE then table.remove(h, 1) end
end

local function getMovementDirectionFromHistory(character)
    if not character or not positionHistory[character] then return Vector3.new(0,0,0) end
    local h = positionHistory[character]
    if #h < 2 then return Vector3.new(0,0,0) end
    local m = h[#h] - h[1]
    local rm = Vector3.new(0,0,0)
    if #h >= 3 then rm = h[#h] - h[#h-1] end
    if m.Magnitude > MOVE_THRESHOLD then return m.Unit
    elseif rm.Magnitude > MOVE_THRESHOLD then return rm.Unit end
    return Vector3.new(0,0,0)
end

local function getInstantaneousDirection(character, currentPos)
    if not character or not currentPos then return Vector3.new(0,0,0) end
    if not positionHistory[character] or #positionHistory[character] < 2 then return Vector3.new(0,0,0) end
    local m = currentPos - positionHistory[character][#positionHistory[character]]
    if m.Magnitude > MOVE_THRESHOLD then return m.Unit end
    return Vector3.new(0,0,0)
end

local function getPredictedPosition(character, multiplier)
    if not character then return nil end
    local root = character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Head") or character:FindFirstChild("Torso")
    if not root then return nil end
    local p = root.Position
    updatePositionHistory(character, p)
    local md = getMovementDirectionFromHistory(character)
    local id = getInstantaneousDirection(character, p)
    local fd = id.Magnitude > MOVE_THRESHOLD and id or md
    local dist = BASE_PREDICTION_DISTANCE * (multiplier or aimbotPredictionMultiplier)
    if fd.Magnitude > MOVE_THRESHOLD then return p + (fd * dist) end
    return p
end

local function createTargetSphere(targetCharacter, sphereColor, predictionMultiplier)
    if currentTargetSphere then pcall(function() currentTargetSphere:Destroy() end); currentTargetSphere = nil end
    if currentSphereConnection then pcall(function() currentSphereConnection:Disconnect() end); currentSphereConnection = nil end
    if not targetCharacter then return nil end
    local root = targetCharacter:FindFirstChild("HumanoidRootPart") or targetCharacter:FindFirstChild("Head") or targetCharacter:FindFirstChild("Torso")
    if not root then return nil end
    local sphere = Instance.new("Part")
    sphere.Shape = Enum.PartType.Ball
    sphere.Size = Vector3.new(2,2,2)
    sphere.Material = Enum.Material.Neon
    sphere.Color = sphereColor or Color3.fromRGB(255,0,0)
    sphere.Anchored = true
    sphere.CanCollide = false
    sphere.Name = "TargetSphere"
    sphere.Parent = workspace
    sphere.Position = root.Position
    updatePositionHistory(targetCharacter, root.Position)
    local conn
    conn = RunService.Heartbeat:Connect(function()
        if not targetCharacter or not targetCharacter.Parent or not sphere.Parent then
            sphere:Destroy(); if conn then conn:Disconnect() end; currentTargetSphere = nil; currentSphereConnection = nil; return
        end
        local hum = targetCharacter:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then
            sphere:Destroy(); if conn then conn:Disconnect() end; currentTargetSphere = nil; currentSphereConnection = nil; return
        end
        root = targetCharacter:FindFirstChild("HumanoidRootPart") or targetCharacter:FindFirstChild("Head") or targetCharacter:FindFirstChild("Torso")
        if not root then
            sphere:Destroy(); if conn then conn:Disconnect() end; currentTargetSphere = nil; currentSphereConnection = nil; return
        end
        local p = root.Position
        updatePositionHistory(targetCharacter, p)
        local md = getMovementDirectionFromHistory(targetCharacter)
        local id = getInstantaneousDirection(targetCharacter, p)
        local fd = id.Magnitude > MOVE_THRESHOLD and id or md
        local mult = predictionMultiplier or aimbotPredictionMultiplier
        local dist = BASE_PREDICTION_DISTANCE * mult
        local target = fd.Magnitude > MOVE_THRESHOLD and (p + (fd * dist)) or p
        sphere.Position = sphere.Position:Lerp(target, SPHERE_SMOOTHNESS)
    end)
    currentTargetSphere = sphere
    currentSphereConnection = conn
    return sphere, conn
end

local function getValidTargets()
    local targets = {}
    local myChar = lp.Character
    if not myChar then return targets end
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return targets end
    local myTeam, myCharacter = getMyCharacterInfo()
    local isElliot = (myTeam == "Survivor" and myCharacter == "Elliot")
    local vp = Camera.ViewportSize
    local center = Vector2.new(vp.X/2, vp.Y/2)
    local function processTarget(obj)
        local u = obj:GetAttribute("Username")
        if u and u == lp.Name then return end
        local hum = obj:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then return end
        local part = obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChild("Head") or obj:FindFirstChild("Torso")
        if not part then return end
        local distance = (part.Position - myRoot.Position).Magnitude
        if distance > aimbotMaxDistance then return end
        if aimbotUseFOV then
            local sp, onScreen = Camera:WorldToViewportPoint(part.Position)
            if onScreen then
                local sv = Vector2.new(sp.X, sp.Y)
                local dfc = (sv - center).Magnitude
                if dfc <= aimbotFOV then
                    table.insert(targets, { obj = obj, part = part, distance = distance, screenDistance = dfc })
                end
            end
        else
            table.insert(targets, { obj = obj, part = part, distance = distance, screenDistance = 0 })
        end
    end
    if isElliot then
        for _, s in ipairs(SurvivorsFolder:GetChildren()) do processTarget(s) end
    elseif myTeam == "Killer" then
        for _, s in ipairs(SurvivorsFolder:GetChildren()) do processTarget(s) end
    elseif myTeam == "Survivor" then
        for _, k in ipairs(KillersFolder:GetChildren()) do processTarget(k) end
    end
    if aimbotUseFOV then
        table.sort(targets, function(a,b) return a.screenDistance < b.screenDistance end)
    else
        table.sort(targets, function(a,b) return a.distance < b.distance end)
    end
    return targets
end

local function getNearestKiller()
    local myChar = lp.Character
    if not myChar then return nil end
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end
    local nearest, nearestDist = nil, math.huge
    for _, k in ipairs(KillersFolder:GetChildren()) do
        local u = k:GetAttribute("Username")
        if not (u and u == lp.Name) then
            local hrp = k:FindFirstChild("HumanoidRootPart")
            if hrp then
                local hum = k:FindFirstChildOfClass("Humanoid")
                if hum and hum.Health > 0 then
                    local d = (hrp.Position - myRoot.Position).Magnitude
                    if d < nearestDist then nearestDist = d; nearest = k end
                end
            end
        end
    end
    return nearest
end

-- ===== M1 WARNING FUNCTIONS =====

function cleanupM1Warning(killer)
    if not killer then return end
    local existingHighlight = killer:FindFirstChild("M1WarningHighlight")
    if existingHighlight then existingHighlight:Destroy() end
    if killer:GetAttribute("M1WarningActive") then
        killer:SetAttribute("M1WarningActive", nil)
    end
    local torso = killer:FindFirstChild("Torso") or killer:FindFirstChild("UpperTorso") or killer:FindFirstChild("HumanoidRootPart")
    if torso then
        local existingBillboard = torso:FindFirstChild("M1WarningBillboard")
        if existingBillboard then existingBillboard:Destroy() end
    end
end

function showM1Warning(killer, duration)
    if not killer then return end
    if not killerM1WarningEnabled then return end
    cleanupM1Warning(killer)
    killer:SetAttribute("M1WarningActive", true)

    -- Steady red highlight
    local highlight = Instance.new("Highlight")
    highlight.Name = "M1WarningHighlight"
    highlight.FillColor = Color3.fromRGB(255, 0, 0)
    highlight.OutlineColor = Color3.fromRGB(255, 0, 0)
    highlight.FillTransparency = 0.3
    highlight.OutlineTransparency = 0
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.Adornee = killer
    highlight.Parent = killer

    -- Exclamation mark on torso
    local torso = killer:FindFirstChild("Torso") or killer:FindFirstChild("UpperTorso") or killer:FindFirstChild("HumanoidRootPart")
    local text = nil
    if torso then
        local billboard = Instance.new("BillboardGui")
        billboard.Name = "M1WarningBillboard"
        billboard.Size = UDim2.new(0, 60, 0, 60)
        billboard.StudsOffset = Vector3.new(0, 0.2, 0)
        billboard.AlwaysOnTop = true
        billboard.Parent = torso

        text = Instance.new("TextLabel")
        text.Size = UDim2.new(1, 0, 1, 0)
        text.BackgroundTransparency = 1
        text.Text = "!"
        text.TextColor3 = Color3.fromRGB(255, 0, 0)
        text.TextStrokeTransparency = 0
        text.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
        text.TextScaled = true
        text.Font = Enum.Font.GothamBlack
        text.Parent = billboard
    end

    -- Alarm-light pulse on the "!" only
    local flashConn
    local t = 0
    local PULSE_SPEED = 4
    flashConn = RunService.Heartbeat:Connect(function(dt)
        if not killer or not killer.Parent or not killer:GetAttribute("M1WarningActive") then
            if flashConn then flashConn:Disconnect() end
            return
        end
        if not text then return end
        t = t + dt * PULSE_SPEED
        local pulse = (math.sin(t * math.pi * 2) + 1) / 2
        text.TextColor3 = Color3.fromRGB(255, 0, 0)
        text.TextTransparency = 0.2 + (1 - pulse) * 0.5
    end)

    -- Auto-remove
    task.delay(duration or 0.5, function()
        if killer then
            killer:SetAttribute("M1WarningActive", nil)
        end
        if flashConn then flashConn:Disconnect() end
        if highlight then pcall(function() highlight:Destroy() end) end
        if torso then
            local b = torso:FindFirstChild("M1WarningBillboard")
            if b then pcall(function() b:Destroy() end) end
        end
    end)
end

-- Animation start-time tracking for M1 Warning
function setupAnimationStartTracking(killer)
    if not killer then return end
    if killer:GetAttribute("AnimStartTracked") then return end
    killer:SetAttribute("AnimStartTracked", true)
    local hum = killer:FindFirstChildOfClass("Humanoid")
    if not hum then
        hum = killer:WaitForChild("Humanoid", 5)
        if not hum then return end
    end
    local function setupAnim(a)
        if not a then return end
        a.AnimationPlayed:Connect(function(track)
            local id = extractNumericAnimationId(track.Animation.AnimationId)
            if id then
                if not animationStartTimes[killer] then animationStartTimes[killer] = {} end
                animationStartTimes[killer][id] = tick()
            end
        end)
    end
    local a = hum:FindFirstChildOfClass("Animator")
    if a then setupAnim(a) end
    hum.ChildAdded:Connect(function(c) if c:IsA("Animator") then setupAnim(c) end end)
end

function scanForAnimationTracking()
    if not KillersFolder then return end
    for _, k in ipairs(KillersFolder:GetChildren()) do
        if not isLocalPlayer(k) then
            setupAnimationStartTracking(k)
        end
    end
end

function watchForNewKillersForAnimTracking()

    KillersFolder.ChildAdded:Connect(function(c)
        task.wait(0.5)
        if not isLocalPlayer(c) then setupAnimationStartTracking(c) end
    end)
    KillersFolder.ChildRemoved:Connect(function(c)
        if animationStartTimes[c] then animationStartTimes[c] = nil end
        cleanupM1Warning(c)
    end)
end

-- ===== SLASHER BEHEAD TRACKING (with casting window) =====
function isSlasherEnragedForBehead(slasher)
    local head = slasher:FindFirstChild("Head")
    if head then
        local rpa = head:FindFirstChild("RagingPaceAttachment")
        if rpa then
            local rp = rpa:FindFirstChild("RagingPace")
            if rp and rp:IsA("ParticleEmitter") then return rp.Enabled == true end
        end
        local rp = head:FindFirstChild("RagingPace")
        if rp then return rp.Enabled == true end
    end
    local plate = slasher:FindFirstChild("Plate")
    if plate then
        local s = plate:FindFirstChild("String")
        if s then return s.Transparency == 1 end
    end
    return false
end

function isSlasherBeheadReady(slasher)
    local endTime = slasherBeheadCooldowns[slasher]
    if not endTime then return true end
    return tick() >= endTime
end

function isSlasherBeheadCasting(slasher)
    return slasherBeheadCasting[slasher] == true
end

function setupSlasherBeheadTracking(slasher)
    if not slasher then return end
    if slasherBeheadTracked[slasher] then return end
    slasherBeheadTracked[slasher] = true
    local hum = slasher:FindFirstChildOfClass("Humanoid")
    if not hum then
        hum = slasher:WaitForChild("Humanoid", 5)
        if not hum then return end
    end
    local function setupAnim(a)
        if not a then return end
        a.AnimationPlayed:Connect(function(track)
            local id = extractNumericAnimationId(track.Animation.AnimationId)
            if not id then return end
            for _, i in ipairs(BEHEAD_ANIMATIONS) do
                if id == i then
                    local enraged = isSlasherEnragedForBehead(slasher)
                    local cd = enraged and SLASHER_BEHEAD_ENRAGED or SLASHER_BEHEAD_NORMAL
                    slasherBeheadCooldowns[slasher] = tick() + cd
                    slasherBeheadCasting[slasher] = true
                    task.delay(2, function()
                        if slasherBeheadCasting[slasher] then
                            slasherBeheadCasting[slasher] = false
                        end
                    end)
                    return
                end
            end
        end)
    end
    local a = hum:FindFirstChildOfClass("Animator")
    if a then setupAnim(a) end
    hum.ChildAdded:Connect(function(c) if c:IsA("Animator") then setupAnim(c) end end)
end

function watchForSlasherBeheadTracking()
    if not KillersFolder then return end
    for _, k in ipairs(KillersFolder:GetChildren()) do
        if k.Name == "Slasher" and not slasherBeheadTracked[k] then setupSlasherBeheadTracking(k) end
    end
    KillersFolder.ChildAdded:Connect(function(c)
        if c.Name == "Slasher" then task.wait(0.5); setupSlasherBeheadTracking(c) end
    end)
    KillersFolder.ChildRemoved:Connect(function(c)
        if c.Name == "Slasher" then
            slasherBeheadCooldowns[c] = nil
            slasherBeheadTracked[c] = nil
            slasherBeheadCasting[c] = nil
        end
    end)
end

function checkForAttackSounds()
    if not autoBlockForGuest then return end
    local now = tick()
    if now - lastBlockTime < BLOCK_COOLDOWN then return end
    local myChar = lp.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    if not KillersFolder then return end
    local isG = amIGuest1337()
    if not isG then return end
    for _, killer in ipairs(KillersFolder:GetChildren()) do
        local u = killer:GetAttribute("Username")
        if not (u and u == lp.Name) then
            local khrp = killer:FindFirstChild("HumanoidRootPart")
            if khrp then
                local dist = (khrp.Position - myRoot.Position).Magnitude
                if dist <= detectionRange then
                    if (not facingCheckEnabled) or isFacing(myRoot, khrp) then
                        if shouldBlock() then
                            local doBlock = false
                            local detectedBy = nil  -- "animation" or "sound"

                            if killer.Name == "Slasher" then
                                local beheadReady = isSlasherBeheadReady(killer)
                                local casting = isSlasherBeheadCasting(killer)
                                if beheadReady or casting then
                                    -- Use animations (start-time tracked)
                                    local hum = killer:FindFirstChildOfClass("Humanoid")
                                    if hum then
                                        local a = hum:FindFirstChildOfClass("Animator")
                                        if a then
                                            local window = KILLER_ATTACK_WINDOWS[killer.Name] or DEFAULT_ATTACK_WINDOW
                                            for _, t in ipairs(a:GetPlayingAnimationTracks()) do
                                                local id = extractNumericAnimationId(t.Animation.AnimationId)
                                                if id and autoBlockTriggerAnimations[id] then
                                                    local startedAt = animationStartTimes[killer] and animationStartTimes[killer][id]
                                                    if startedAt and (tick() - startedAt) < window then
                                                        doBlock = true
                                                        detectedBy = "animation"
                                                        break
                                                    end
                                                end
                                            end
                                        end
                                    end
                                else
                                    -- Sound check
                                    local window = KILLER_ATTACK_WINDOWS[killer.Name] or DEFAULT_ATTACK_WINDOW
                                    for _, s in ipairs(killer:GetDescendants()) do
                                        if s:IsA("Sound") then
                                            local id = extractNumericSoundId(s)
                                            if id and autoBlockTriggerSounds[id] then
                                                if s.TimePosition > 0 and s.TimePosition < window then
                                                    doBlock = true
                                                    detectedBy = "sound"
                                                    break
                                                end
                                            end
                                        end
                                    end
                                end
                            else
                                -- Non-Slasher: sound check
                                local window
                                if killer.Name == "Sixer" then
                                    window = SIXER_SOUND_WINDOW
                                else
                                    window = KILLER_ATTACK_WINDOWS[killer.Name] or DEFAULT_ATTACK_WINDOW
                                end
                                for _, s in ipairs(killer:GetDescendants()) do
                                    if s:IsA("Sound") then
                                        local id = extractNumericSoundId(s)
                                        if id and autoBlockTriggerSounds[id] then
                                            if s.TimePosition > 0 and s.TimePosition < window then
                                                doBlock = true
                                                detectedBy = "sound"
                                                break
                                            end
                                        end
                                    end
                                end
                            end

                            if doBlock then
                                -- Trigger M1 warning
                                if killerM1WarningEnabled then
                                    local w = KILLER_ATTACK_WINDOWS[killer.Name] or DEFAULT_ATTACK_WINDOW
                                    if killer.Name == "Sixer" then w = 0.3 end
                                    showM1Warning(killer, w)
                                end

                                if blockdelay > 0 then task.wait(blockdelay) end
                                refreshUIRefs()
                                if isHelpless() then break end
                                if autoblocktype == "Block" then
                                    if cachedCooldown and cachedCooldown.Text == "" then
                                        pressKey(Enum.KeyCode.Q)
                                        lastBlockTime = now
                                        return
                                    end
                                elseif autoblocktype == "Charge" then
                                    if cachedChargeBtn and cachedChargeBtn:FindFirstChild("CooldownTime") and cachedChargeBtn.CooldownTime.Text == "" then
                                        pressKey(Enum.KeyCode.Q)
                                        lastBlockTime = now
                                        return
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end
end

function checkSpecialKillerStatus()
    local character = lp.Character
    if not character then
        if isPlayingAsSpecialKiller then
            isPlayingAsSpecialKiller = false
            currentKillerType = nil
            facingCheckEnabled = originalFacingCheckState
        end
        return false
    end
    for _, s in ipairs(getAllCharactersByName("Slasher")) do
        local u = s:GetAttribute("Username")
        if u and u == lp.Name then
            if not isPlayingAsSpecialKiller or currentKillerType ~= "Slasher" then
                isPlayingAsSpecialKiller = true
                currentKillerType = "Slasher"
                originalFacingCheckState = facingCheckEnabled
                facingCheckEnabled = false
            end
            return true
        end
    end
    for _, j in ipairs(getAllCharactersByName("JohnDoe")) do
        local u = j:GetAttribute("Username")
        if u and u == lp.Name then
            if not isPlayingAsSpecialKiller or currentKillerType ~= "JohnDoe" then
                isPlayingAsSpecialKiller = true
                currentKillerType = "JohnDoe"
                originalFacingCheckState = facingCheckEnabled
                facingCheckEnabled = false
            end
            return true
        end
    end
    for _, o in ipairs(getAllCharactersByName("1x1x1x1")) do
        local u = o:GetAttribute("Username")
        if u and u == lp.Name then
            if not isPlayingAsSpecialKiller or currentKillerType ~= "1x1x1x1" then
                isPlayingAsSpecialKiller = true
                currentKillerType = "1x1x1x1"
                originalFacingCheckState = facingCheckEnabled
                facingCheckEnabled = false
            end
            return true
        end
    end
    if isPlayingAsSpecialKiller then
        isPlayingAsSpecialKiller = false
        currentKillerType = nil
        facingCheckEnabled = originalFacingCheckState
    end
    return false
end

-- ===== SMOOTH SHIFTLOCK =====
SmoothShiftLock = nil
shiftlockLoaded = false
shiftlockWasEnabled = false

function loadSmoothShiftLock()
    local s, m = pcall(function()
        return require(ReplicatedStorage:FindFirstChild("Systems"):FindFirstChild("Player"):FindFirstChild("Game"):FindFirstChild("SmoothShiftLock"))
    end)
    if not s then
        s, m = pcall(function() return require(ReplicatedStorage.Systems.Player.Game.SmoothShiftLock) end)
    end
    if s and m then
        SmoothShiftLock = m
        shiftlockLoaded = true
    else
        shiftlockLoaded = false
    end
end
loadSmoothShiftLock()

function isShiftLockEnabled()
    if shiftlockLoaded and SmoothShiftLock then return SmoothShiftLock.Enabled end
    return UserInputService.MouseBehavior == Enum.MouseBehavior.LockCenter
end

function setShiftLock(enabled, fromAimbot)
    if enabled then
        if fromAimbot then
            shiftlockWasEnabled = isShiftLockEnabled()
            shiftlockWasAimbotActive = true
        end
        if not isShiftLockEnabled() then
            if shiftlockLoaded and SmoothShiftLock and not SmoothShiftLock.Enabled then
                pcall(function() SmoothShiftLock:ToggleShiftLock() end)
            else
                pcall(function()
                    local sk = lp.PlayerData.Settings.Keybinds.ShiftLock.Value
                    local kc = Enum.KeyCode[sk]
                    if kc then
                        VirtualInputManager:SendKeyEvent(true, kc, false, game)
                        VirtualInputManager:SendKeyEvent(false, kc, false, game)
                    end
                end)
            end
        end
    else
        if fromAimbot and shiftlockWasAimbotActive then
            if not shiftlockWasEnabled then
                if isShiftLockEnabled() then
                    if shiftlockLoaded and SmoothShiftLock and SmoothShiftLock.Enabled then
                        pcall(function() SmoothShiftLock:ToggleShiftLock() end)
                    else
                        pcall(function()
                            local sk = lp.PlayerData.Settings.Keybinds.ShiftLock.Value
                            local kc = Enum.KeyCode[sk]
                            if kc then
                                VirtualInputManager:SendKeyEvent(true, kc, false, game)
                                VirtualInputManager:SendKeyEvent(false, kc, false, game)
                            end
                        end)
                    end
                end
            end
            shiftlockWasAimbotActive = false
        end
    end
end

-- ===== CHARACTER LISTENER CLEANUP =====
function cleanupAbilityTimers(timerTable, character)
    if timerTable[character] then
        for _, data in pairs(timerTable[character]) do
            if data.billboard then pcall(function() data.billboard:Destroy() end) end
            if data.connection then data.connection:Disconnect() end
        end
        timerTable[character] = nil
    end
end

function destroyNoobTimer(timerData)
    if not timerData then return end
    if timerData.billboard then pcall(timerData.billboard.Destroy, timerData.billboard) end
    if timerData.connection then timerData.connection:Disconnect() end
end

function cleanupCharacterTimers(character)
    if positionHistory[character] then positionHistory[character] = nil end
    if animationStartTimes[character] then animationStartTimes[character] = nil end
    if aimbotCurrentTarget == character then stopAimbot() end
    if currentDusekkarTarget == character then stopDusekkar() end
    if activeCooldownTimers[character] then
        if activeCooldownTimers[character].billboard then pcall(function() activeCooldownTimers[character].billboard:Destroy() end) end
        if activeCooldownTimers[character].connection then activeCooldownTimers[character].connection:Disconnect() end
        activeCooldownTimers[character] = nil
    end
    cleanupAbilityTimers(slasherActiveCooldownTimers, character)
    cleanupAbilityTimers(onexActiveCooldownTimers, character)
    cleanupAbilityTimers(sixerActiveCooldownTimers, character)
    cleanupAbilityTimers(noliActiveCooldownTimers, character)
    cleanupAbilityTimers(c00lkiddActiveCooldownTimers, character)
    cleanupAbilityTimers(johnDoeActiveCooldownTimers, character)
    cleanupAbilityTimers(nosferatuActiveCooldownTimers, character)
    cleanupAbilityTimers(azureActiveCooldownTimers, character)
    if noobActiveCooldownTimers[character] then
        for _, data in pairs(noobActiveCooldownTimers[character]) do
            destroyNoobTimer(data)
        end
        noobActiveCooldownTimers[character] = nil
    end
    if slasherBeheadCooldowns[character] then slasherBeheadCooldowns[character] = nil end
    if slasherBeheadTracked[character] then slasherBeheadTracked[character] = nil end
    if slasherBeheadCasting[character] then slasherBeheadCasting[character] = nil end
    cleanupM1Warning(character)
end

function watchCharacterRemoval(folder)
    if not folder then return end
    folder.ChildRemoved:Connect(function(c) cleanupCharacterTimers(c) end)
end

function initCharacterListeners()
    if SurvivorsFolder then SurvivorsFolder.ChildAdded:Connect(setupSurvivorListener) end
    if KillersFolder then KillersFolder.ChildAdded:Connect(setupKillerListener) end
    watchCharacterRemoval(KillersFolder)
    watchCharacterRemoval(SurvivorsFolder)
end

function cleanupNoliAimbot()
    if not noliAimbotActive then return end
    noliAimbotActive = false
    if noliAimbotCameraConnection then noliAimbotCameraConnection:Disconnect(); noliAimbotCameraConnection = nil end
    for _, conn in ipairs(noliAimbotSoundConnections) do conn:Disconnect() end
    noliAimbotSoundConnections = {}
    aimbotActive = false
    aimbotCurrentTarget = nil
    if currentTargetSphere then pcall(function() currentTargetSphere:Destroy() end); currentTargetSphere = nil end
    if currentSphereConnection then pcall(function() currentSphereConnection:Disconnect() end); currentSphereConnection = nil end
    setShiftLock(false, true)
end

function stopNoliAimbot()
    if not noliAimbotActive then return end
    if noliAimbotTimer then pcall(function() task.cancel(noliAimbotTimer) end); noliAimbotTimer = nil end
    cleanupNoliAimbot()
end

function setupNoliSoundHook()
    local h = lp.PlayerData.Settings.Customization:FindFirstChild("HitsoundID")
    local k = lp.PlayerData.Settings.Customization:FindFirstChild("KillsoundID")
    if not h or not k then return false end
    noliHitID = h.Value
    noliKillID = k.Value
    local s, m = pcall(function() return require(ReplicatedStorage.Modules.Rendering.Sounds) end)
    if not s then
        s, m = pcall(function() return require(ReplicatedStorage:FindFirstChild("Modules"):FindFirstChild("Rendering"):FindFirstChild("Sounds")) end)
    end
    if not s or not m then noliSoundsModuleHooked = false; return false end
    noliSoundsModule = m
    if not noliSoundsModuleHooked then
        noliSoundsModuleHooked = true
        noliOriginalSoundsPlay = m.Play
        m.Play = function(self, soundId, ...)
            local args = {...}
            local id = tostring(soundId)
            local num = id:match("rbxassetid://(%d+)") or id:match("://(%d+)") or id:match("^(%d+)$")
            if num == noliHitID or num == noliKillID then
                if noliAimbotActive then stopNoliAimbot() end
            end
            return noliOriginalSoundsPlay(self, soundId, unpack(args))
        end
    end
    return true
end

function startNoliAimbot()
    if not aimbotEnabled then return end
    if noliAimbotActive then return end
    stopAimbot(); stopNoliAimbot()
    setupNoliSoundHook()
    local targets = getValidTargets()
    if #targets == 0 then return end
    local target = targets[1]
    aimbotActive = true
    aimbotCurrentTarget = target.obj
    noliAimbotActive = true
    setShiftLock(true, true)
    createTargetSphere(aimbotCurrentTarget, Color3.fromRGB(255, 0, 0), aimbotPredictionMultiplier)
    local conn
    conn = RunService.RenderStepped:Connect(function()
        if not aimbotActive or not aimbotCurrentTarget then
            if conn then conn:Disconnect() end
            stopNoliAimbot()
            return
        end
        local hum = aimbotCurrentTarget:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then
            stopNoliAimbot(); if conn then conn:Disconnect() end; return
        end
        local tp = getPredictedPosition(aimbotCurrentTarget, aimbotPredictionMultiplier)
        if not tp then stopNoliAimbot(); if conn then conn:Disconnect() end; return end
        local cp = Camera.CFrame.Position
        local targetLook = CFrame.lookAt(cp, tp).LookVector
        local smoothLook = Camera.CFrame.LookVector:Lerp(targetLook, aimbotSmoothness)
        Camera.CFrame = CFrame.new(cp, cp + smoothLook)
    end)
    noliAimbotCameraConnection = conn
    local cac = lp.CharacterAdded:Connect(function()
        if noliAimbotActive then stopNoliAimbot() end
    end)
    table.insert(noliAimbotSoundConnections, cac)
    if noliAimbotTimer then task.cancel(noliAimbotTimer); noliAimbotTimer = nil end
    noliAimbotTimer = task.delay(3, function()
        if noliAimbotActive then cleanupNoliAimbot() end
    end)
end

function startNormalStab(targetKiller)
    if not targetKiller then return end
    local char = lp.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    if faceSameSideAsKiller then
        setShiftLock(false, false)
        createTargetSphere(targetKiller, Color3.fromRGB(255,0,0))
        local bsConn
        bsConn = RunService.Heartbeat:Connect(function()
            if not aimbotActive then if bsConn then bsConn:Disconnect() end; return end
            if not targetKiller or not targetKiller.Parent then stopAimbot(); if bsConn then bsConn:Disconnect() end; return end
            local khrp = targetKiller:FindFirstChild("HumanoidRootPart")
            if khrp then
                local kl = khrp.CFrame.LookVector
                if kl.Magnitude > 0 then hrp.CFrame = CFrame.lookAt(hrp.Position, hrp.Position + kl) end
            end
        end)
        if aimbotActiveTimer then pcall(function() task.cancel(aimbotActiveTimer) end) end
        aimbotActiveTimer = task.delay(0.4, function()
            aimbotActive = false; aimbotCurrentTarget = nil
            if bsConn then bsConn:Disconnect() end
            if currentTargetSphere then pcall(function() currentTargetSphere:Destroy() end); currentTargetSphere = nil end
        end)
    else
        setShiftLock(true, false)
        createTargetSphere(targetKiller, Color3.fromRGB(255,0,0))
        local cc
        cc = RunService.RenderStepped:Connect(function()
            if not aimbotActive then if cc then cc:Disconnect() end; return end
            if not targetKiller or not targetKiller.Parent then stopAimbot(); if cc then cc:Disconnect() end; return end
            local tp
            if currentTargetSphere and currentTargetSphere.Parent then tp = currentTargetSphere.Position
            else
                local khrp = targetKiller:FindFirstChild("HumanoidRootPart")
                if khrp then tp = khrp.Position else stopAimbot(); if cc then cc:Disconnect() end; return end
            end
            local cp = Camera.CFrame.Position
            local targetLook = CFrame.lookAt(cp, tp).LookVector
            local smoothLook = Camera.CFrame.LookVector:Lerp(targetLook, aimbotSmoothness)
            Camera.CFrame = CFrame.new(cp, cp + smoothLook)
        end)
        if aimbotActiveTimer then pcall(function() task.cancel(aimbotActiveTimer) end) end
        aimbotActiveTimer = task.delay(0.4, function()
            aimbotActive = false; aimbotCurrentTarget = nil
            if cc then cc:Disconnect() end
            if currentTargetSphere then pcall(function() currentTargetSphere:Destroy() end); currentTargetSphere = nil end
            setShiftLock(false, false)
        end)
    end
end

function startGeneralAimbot(duration, customSmoothness)
    if not aimbotEnabled then return end
    if aimbotActive then return end
    local targets = getValidTargets()
    if #targets == 0 then return end
    local target = targets[1]
    aimbotCurrentTarget = target.obj
    aimbotActive = true
    aimbotTargetCount = aimbotTargetCount + 1
    setShiftLock(true, true)
    local origSmooth = aimbotSmoothness
    if customSmoothness then aimbotSmoothness = customSmoothness end
    createTargetSphere(aimbotCurrentTarget, Color3.fromRGB(255,0,0), aimbotPredictionMultiplier)
    local cc
    cc = RunService.RenderStepped:Connect(function()
        if not aimbotActive or not aimbotCurrentTarget then if cc then cc:Disconnect() end; return end
        local hum = aimbotCurrentTarget:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then stopAimbot(); if cc then cc:Disconnect() end; return end
        local tp = getPredictedPosition(aimbotCurrentTarget, aimbotPredictionMultiplier)
        if not tp then stopAimbot(); if cc then cc:Disconnect() end; return end
        local cp = Camera.CFrame.Position
        local targetLook = CFrame.lookAt(cp, tp).LookVector
        local smoothLook = Camera.CFrame.LookVector:Lerp(targetLook, aimbotSmoothness)
        Camera.CFrame = CFrame.new(cp, cp + smoothLook)
    end)
    if aimbotActiveTimer then pcall(function() task.cancel(aimbotActiveTimer) end) end
    aimbotActiveTimer = task.delay(duration, function()
        if customSmoothness then aimbotSmoothness = origSmooth end
        aimbotActive = false; aimbotCurrentTarget = nil
        if cc then cc:Disconnect() end
        if currentTargetSphere then pcall(function() currentTargetSphere:Destroy() end); currentTargetSphere = nil end
        setShiftLock(false, true)
    end)
end

-- ===== BLOOD ORB =====
function bringBloodOrbsToPlayer()
    if not autoBloodOrb then return end
    if not amISixer() then return end
    local bui = PlayerGui:FindFirstChild("TemporaryUI")
    if bui then bui = bui:FindFirstChild("BloodUI") end
    if bui then
        local b = bui:FindFirstChild("Blood")
        local t = bui:FindFirstChild("Timer")
        if b then
            local c = b:FindFirstChild("Count")
            local m = b:FindFirstChild("Max")
            if c and m then
                local cc = c:FindFirstChild("ContentText")
                local mc = m:FindFirstChild("ContextText")
                if cc and mc and cc.Text == mc.Text then return end
            end
        end
        if t then
            local tt = t:FindFirstChild("Time")
            if tt then
                local ct = tt:FindFirstChild("ContentText")
                if ct then
                    local tv = ct.Text
                    if tv ~= "0" and tv ~= "TIME'S UP" then return end
                end
            end
        end
    end
    local boc = Workspace:FindFirstChild("Map")
    if boc then boc = boc:FindFirstChild("vrrebluxBloodOrbContainer") end
    if not boc then return end
    local ch = lp.Character
    if not ch then return end
    local hrp = ch:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    for _, orb in ipairs(boc:GetChildren()) do
        if orb.Name == "BloodOrb" and orb:IsA("BasePart") then
            pcall(function()
                local tw = TweenService:Create(orb, TweenInfo.new(0.3, Enum.EasingStyle.Linear), {Position = hrp.Position})
                tw:Play()
                tw.Completed:Wait()
            end)
        end
    end
end

function startBloodOrbLoop()
    if bloodOrbConnection then bloodOrbConnection:Disconnect(); bloodOrbConnection = nil end
    if not autoBloodOrb then return end
    bloodOrbConnection = RunService.Heartbeat:Connect(function() pcall(bringBloodOrbsToPlayer) end)
end

-- ===== NOSFERATU TRACKER =====
function getAllNosferatus()
    local out = {}
    local pf = Workspace:FindFirstChild("Players")
    if pf then
        local kf = pf:FindFirstChild("Killers")
        if kf then
            for _, k in ipairs(kf:GetChildren()) do
                if k.Name == "Nosferatu" then table.insert(out, k) end
            end
        end
    end
    return out
end

function formatNosferatuCooldownTime(s)
    if s <= 0 then return "0.0"
    elseif s < 10 then return string.format("%.1f", s)
    else return tostring(math.floor(s)) end
end

function createNosferatuCooldownDisplay(n, endTime, ability)
    if nosferatuActiveCooldownTimers[n] and nosferatuActiveCooldownTimers[n][ability] then
        local ex = nosferatuActiveCooldownTimers[n][ability]
        if ex.billboard then pcall(function() ex.billboard:Destroy() end) end
        if ex.connection then ex.connection:Disconnect() end
        nosferatuActiveCooldownTimers[n][ability] = nil
    end
    local hrp = n:FindFirstChild("HumanoidRootPart") or n:FindFirstChild("Head") or n:FindFirstChild("Torso")
    if not hrp then return end
    local oy = NOSFERATU_ABILITY_OFFSETS[ability] or 3
    local bb = Instance.new("BillboardGui")
    bb.Name = "NosferatuCooldown_" .. ability
    bb.Size = UDim2.new(0, 140, 0, 25)
    bb.StudsOffset = Vector3.new(0, oy, 0)
    bb.AlwaysOnTop = true
    bb.Parent = hrp
    local tl = Instance.new("TextLabel")
    tl.Size = UDim2.new(1, 0, 1, 0)
    tl.BackgroundTransparency = 1
    tl.Text = ability .. ": " .. formatNosferatuCooldownTime(endTime - tick())
    tl.TextStrokeTransparency = 0
    tl.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    tl.TextScaled = true
    tl.Font = Enum.Font.GothamBold
    tl.Parent = bb
    if not nosferatuActiveCooldownTimers[n] then nosferatuActiveCooldownTimers[n] = {} end
    local td = { endTime = endTime, billboard = bb, textLabel = tl, abilityName = ability, connection = nil }
    local conn
    conn = RunService.Heartbeat:Connect(function()
        if not n or not n.Parent then
            if conn then conn:Disconnect() end
            if bb then pcall(function() bb:Destroy() end) end
            if nosferatuActiveCooldownTimers[n] then nosferatuActiveCooldownTimers[n][ability] = nil end
            return
        end
        local rem = td.endTime - tick()
        if rem <= 0.1 then
            if conn then conn:Disconnect() end
            if bb then pcall(function() bb:Destroy() end) end
            if nosferatuActiveCooldownTimers[n] then nosferatuActiveCooldownTimers[n][ability] = nil end
        else
            tl.Text = ability .. ": " .. formatNosferatuCooldownTime(rem)
            if rem < 5 then tl.TextColor3 = Color3.fromRGB(255, 0, 0)
            elseif rem < 10 then tl.TextColor3 = Color3.fromRGB(255, 165, 0)
            else tl.TextColor3 = Color3.fromRGB(255, 255, 255) end
        end
    end)
    td.connection = conn
    nosferatuActiveCooldownTimers[n][ability] = td
end

function resetNosferatuBloodhookCooldown(n)
    if not n then return end
    if nosferatuActiveCooldownTimers[n] and nosferatuActiveCooldownTimers[n]["Bloodhook"] then
        local ex = nosferatuActiveCooldownTimers[n]["Bloodhook"]
        if ex.billboard then pcall(function() ex.billboard:Destroy() end) end
        if ex.connection then ex.connection:Disconnect() end
        nosferatuActiveCooldownTimers[n]["Bloodhook"] = nil
    end
    createNosferatuCooldownDisplay(n, tick() + BLOODHOOK_COOLDOWN, "Bloodhook")
end

function checkForNosferatuBats()
    local m = Workspace:FindFirstChild("Map")
    if not m then return end
    local ig = m:FindFirstChild("Ingame")
    if not ig then return end
    local bats = ig:FindFirstChild("Bats")
    if bats then
        local found
        for _, n in ipairs(getAllNosferatus()) do
            if not isLocalPlayer(n) then
                local hrp = n:FindFirstChild("HumanoidRootPart")
                if hrp and (hrp.Position - bats.Position).Magnitude < 50 then found = n; break end
            end
        end
        if found then
            nosferatuHunterFeastActive[found] = true
            if not nosferatuHunterFeastTracked[found] then
                nosferatuHunterFeastTracked[found] = true
                local conn
                conn = bats.AncestryChanged:Connect(function()
                    if not bats or not bats.Parent then
                        if conn then conn:Disconnect() end
                        nosferatuHunterFeastTracked[found] = nil
                        if nosferatuHunterFeastActive[found] then
                            nosferatuHunterFeastActive[found] = nil
                            createNosferatuCooldownDisplay(found, tick() + HUNTERS_FEAST_COOLDOWN, "Hunter's Feast")
                        end
                    end
                end)
            end
        end
    end
end

function listenForNosferatuAbilities(n)
    if not n then return end
    if isLocalPlayer(n) then return end
    if n:GetAttribute("NosferatuListener") then return end
    n:SetAttribute("NosferatuListener", true)
    local hum = n:FindFirstChildOfClass("Humanoid")
    if hum then
        local function setupAnim(anim)
            if not anim then return end
            anim.AnimationPlayed:Connect(function(track)
                if not nosferatuTrackerEnabled then return end
                local id = extractNumericAnimationId(track.Animation.AnimationId)
                if id == BLOODHOOK_ANIMATION then
                    createNosferatuCooldownDisplay(n, tick() + BLOODHOOK_COOLDOWN, "Bloodhook")
                elseif id == BLOODHOOK_RESET_ANIMATION then
                    resetNosferatuBloodhookCooldown(n)
                elseif id == CATACLYSM_ANIMATION then
                    createNosferatuCooldownDisplay(n, tick() + CATACLYSM_COOLDOWN, "Cataclysm")
                elseif id == ASCENSION_ANIMATION then
                    createNosferatuCooldownDisplay(n, tick() + ASCENSION_COOLDOWN, "Ascension")
                end
            end)
        end
        local a = hum:FindFirstChildOfClass("Animator")
        if a then setupAnim(a) end
        hum.ChildAdded:Connect(function(c) if c:IsA("Animator") then setupAnim(c) end end)
    end
end

function scanForAllNosferatus()
    for _, n in ipairs(getAllNosferatus()) do
        if not isLocalPlayer(n) then listenForNosferatuAbilities(n) end
    end
end

function watchForNewNosferatus()
    local pf = Workspace:FindFirstChild("Players")
    if not pf then return end
    local kf = pf:FindFirstChild("Killers")
    if not kf then return end
    kf.ChildAdded:Connect(function(c)
        if c.Name == "Nosferatu" then
            task.wait(0.5)
            if not isLocalPlayer(c) then listenForNosferatuAbilities(c) end
        end
    end)
end

function startNosferatuTracking()
    if nosferatuUpdateConnection then return end
    scanForAllNosferatus()
    watchForNewNosferatus()
    nosferatuUpdateConnection = RunService.Heartbeat:Connect(function()
        if nosferatuTrackerEnabled then
            scanForAllNosferatus()
            checkForNosferatuBats()
        end
    end)
end

function stopNosferatuTracking()
    if nosferatuUpdateConnection then nosferatuUpdateConnection:Disconnect(); nosferatuUpdateConnection = nil end
    for n, ab in pairs(nosferatuActiveCooldownTimers) do
        for _, d in pairs(ab) do
            if d.billboard then pcall(function() d.billboard:Destroy() end) end
            if d.connection then d.connection:Disconnect() end
        end
        if n then n:SetAttribute("NosferatuListener", nil) end
    end
    nosferatuActiveCooldownTimers = {}
    nosferatuHunterFeastActive = {}
    nosferatuHunterFeastTracked = {}
end

function createNoobCooldownDisplay(noob, endTime, ability)

    if isLocalPlayer(noob) then return end
    if not noobActiveCooldownTimers[noob] then noobActiveCooldownTimers[noob] = {} end
    local ex = noobActiveCooldownTimers[noob][ability]
    if ex then destroyNoobTimer(ex); noobActiveCooldownTimers[noob][ability] = nil end
    local hrp = noob:FindFirstChild("HumanoidRootPart") or noob:FindFirstChild("Head") or noob:FindFirstChild("Torso")
    if not hrp then return end
    local oy = 3
    if ability == "Slateskin Active" or ability == "Slateskin Potion" then oy = 5
    elseif ability == "Ghostburger" then oy = 7 end
    local bb = Instance.new("BillboardGui")
    bb.Name = "NoobCooldown_" .. ability
    bb.Size = UDim2.new(0, 140, 0, 25)
    bb.StudsOffset = Vector3.new(0, oy, 0)
    bb.AlwaysOnTop = true
    bb.Parent = hrp
    local tl = Instance.new("TextLabel")
    tl.Size = UDim2.new(1, 0, 1, 0)
    tl.BackgroundTransparency = 1
    tl.Text = ability .. ": " .. formatNoobCooldownTime(endTime - tick())
    tl.TextStrokeTransparency = 0
    tl.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    tl.TextScaled = true
    tl.Font = Enum.Font.GothamBold
    tl.Parent = bb
    local td = { endTime = endTime, billboard = bb, textLabel = tl, abilityName = ability }
    local conn
    conn = RunService.Heartbeat:Connect(function()
        if not noob or not noob.Parent then
            destroyNoobTimer(td)
            if noobActiveCooldownTimers[noob] then noobActiveCooldownTimers[noob][ability] = nil end
            return
        end
        local rem = td.endTime - tick()
        if rem <= 0 then
            destroyNoobTimer(td)
            if noobActiveCooldownTimers[noob] then noobActiveCooldownTimers[noob][ability] = nil end
        else
            tl.Text = ability .. ": " .. formatNoobCooldownTime(rem)
            if rem < 5 then tl.TextColor3 = Color3.fromRGB(255, 0, 0)
            elseif rem < 10 then tl.TextColor3 = Color3.fromRGB(255, 165, 0)
            else tl.TextColor3 = Color3.fromRGB(255, 255, 255) end
        end
    end)
    td.connection = conn
    noobActiveCooldownTimers[noob][ability] = td
end

    createNoobCooldownDisplay(noob, tick() + NOOB_SLATESKIN_ACTIVE_DURATION, "Slateskin Active")
    noobSlateskinTimerRunning = true
end

function startNoobSlateskinCooldown(noob)
    if isLocalPlayer(noob) then return end
    if noobActiveCooldownTimers[noob] and noobActiveCooldownTimers[noob]["Slateskin Active"] then
        destroyNoobTimer(noobActiveCooldownTimers[noob]["Slateskin Active"])
        noobActiveCooldownTimers[noob]["Slateskin Active"] = nil
    end
    noobSlateskinTimerRunning = false
    local now = tick()
    if not noobLastTriggerTime["SlateskinPotion"] or now - noobLastTriggerTime["SlateskinPotion"] >= NOOB_ANTI_SPAM_TIME then
        noobLastTriggerTime["SlateskinPotion"] = now
        createNoobCooldownDisplay(noob, now + NOOB_SLATESKIN_COOLDOWN, "Slateskin Potion")
    end
end

function checkNoobSlateskinStatus(noob)
    if not noob or not noob.Parent then return end
    local torso = noob:FindFirstChild("Torso")
    if not torso then return end
    local target = BrickColor.new("Dark stone grey")
    local isNow = (torso.BrickColor == target)
    if noobSlateskinActive and not isNow then
        noobSlateskinActive = false
        if noobSlateskinTimerRunning then startNoobSlateskinCooldown(noob) end
        noobSlateskinTimerRunning = false
    end
    if not noobSlateskinActive and isNow then
        noobSlateskinActive = true
        startNoobSlateskinActiveTimer(noob)
    end
end

function listenForNoobAbilities(noob)
    if not noob then return end
    if isLocalPlayer(noob) then return end
    if noob:GetAttribute("NoobListener") then return end
    noob:SetAttribute("NoobListener", true)
    local hum = noob:FindFirstChildOfClass("Humanoid")
    if not hum then
        hum = noob:WaitForChild("Humanoid", 5)
        if not hum then return end
    end
    local function setupAnim(anim)
        if not anim then return end
        anim.AnimationPlayed:Connect(function(track)
            if not cooldownTrackerEnabled then return end
            local id = extractNumericAnimationId(track.Animation.AnimationId)
            if not id then return end
            local now = tick()
            for _, i in ipairs(NOOB_BLOXYCOLA_ANIMATIONS) do
                if id == i then
                    if not noobLastTriggerTime["BloxyCola"] or now - noobLastTriggerTime["BloxyCola"] >= NOOB_ANTI_SPAM_TIME then
                        noobLastTriggerTime["BloxyCola"] = now
                        createNoobCooldownDisplay(noob, now + NOOB_BLOXYCOLA_COOLDOWN, "Bloxy Cola")
                    end
                    return
                end
            end
            for _, i in ipairs(NOOB_SLATESKIN_ANIMATIONS) do
                if id == i then
                    local torso = noob:FindFirstChild("Torso")
                    if torso then
                        noobSlateskinActive = (torso.BrickColor == BrickColor.new("Dark stone grey"))
                        noobSlateskinWasActive = noobSlateskinActive
                    end
                    return
                end
            end
            for _, i in ipairs(NOOB_GHOSTBURGER_ANIMATIONS) do
                if id == i then
                    if not noobLastTriggerTime["Ghostburger"] or now - noobLastTriggerTime["Ghostburger"] >= NOOB_ANTI_SPAM_TIME then
                        noobLastTriggerTime["Ghostburger"] = now
                        createNoobCooldownDisplay(noob, now + NOOB_GHOSTBURGER_COOLDOWN, "Ghostburger")
                    end
                    return
                end
            end
        end)
    end
    local a = hum:FindFirstChildOfClass("Animator")
    if a then setupAnim(a)
    else hum.ChildAdded:Connect(function(c) if c:IsA("Animator") then setupAnim(c) end end) end
end

function scanForAllNoobs()
    local noob = getNoobCharacter()
    if noob and not isLocalPlayer(noob) and not noob:GetAttribute("NoobListener") then
        listenForNoobAbilities(noob)
        local torso = noob:FindFirstChild("Torso")
        if torso then
            noobSlateskinActive = (torso.BrickColor == BrickColor.new("Dark stone grey"))
            noobSlateskinWasActive = noobSlateskinActive
            if noobSlateskinActive then startNoobSlateskinActiveTimer(noob) end
        end
    end
end

function watchForNewNoobs()
    local pf = Workspace:FindFirstChild("Players")
    if not pf then return end
    local sf = pf:FindFirstChild("Survivors")
    if not sf then return end
    sf.ChildAdded:Connect(function(child)
        if child.Name == "Noob" then
            task.wait(0.5)
            if not isLocalPlayer(child) then
                listenForNoobAbilities(child)
                task.wait(0.5)
                local torso = child:FindFirstChild("Torso")
                if torso then
                    noobSlateskinActive = (torso.BrickColor == BrickColor.new("Dark stone grey"))
                    noobSlateskinWasActive = noobSlateskinActive
                    if noobSlateskinActive then startNoobSlateskinActiveTimer(child) end
                end
            end
        end
    end)
end

function startNoobSlateskinLoop()
    if noobSlateskinConnection then return end
    noobSlateskinConnection = RunService.Heartbeat:Connect(function()
        if not cooldownTrackerEnabled then return end
        local noob = getNoobCharacter()
        if noob and not isLocalPlayer(noob) then
            checkNoobSlateskinStatus(noob)
        else
            noobSlateskinActive = false
            noobSlateskinTimerRunning = false
        end
    end)
end

function stopNoobSlateskinLoop()
    if noobSlateskinConnection then noobSlateskinConnection:Disconnect(); noobSlateskinConnection = nil end
end

function startNoobTracking()
    if noobUpdateConnection then return end
    scanForAllNoobs()
    watchForNewNoobs()
    startNoobSlateskinLoop()
    noobUpdateConnection = RunService.Heartbeat:Connect(function()
        if cooldownTrackerEnabled then scanForAllNoobs() end
    end)
end

function stopNoobTracking()
    if noobUpdateConnection then noobUpdateConnection:Disconnect(); noobUpdateConnection = nil end
    stopNoobSlateskinLoop()
    for noob, ab in pairs(noobActiveCooldownTimers) do
        for _, d in pairs(ab) do destroyNoobTimer(d) end
        if noob then noob:SetAttribute("NoobListener", nil) end
    end
    noobActiveCooldownTimers = {}
    noobSlateskinActive = false
    noobSlateskinWasActive = false
    noobSlateskinTimerRunning = false
end

-- ===== AUTO AZURE PUZZLE =====
function doClick() pcall(function() mouse1click() end) end

function getQTE()
    local t = lp.PlayerGui:FindFirstChild("TemporaryUI")
    if not t then return nil end
    return t:FindFirstChild("QTE")
end

function getVisibleZones(qte)
    if not qte then return {} end
    local out = {}
    for i = 1, 3 do
        local z = qte:FindFirstChild("Zone" .. i)
        if z and z.Visible == true then table.insert(out, z) end
    end
    return out
end

function isLineOverZone(line, zone)
    if not line or not zone then return false end
    local lp_, ls = line.AbsolutePosition, line.AbsoluteSize
    local zp, zs = zone.AbsolutePosition, zone.AbsoluteSize
    if not lp_ or not ls or not zp or not zs then return false end
    local lcx = lp_.X + ls.X / 2
    local lcy = lp_.Y + ls.Y / 2
    return (lcx >= zp.X and lcx <= zp.X + zs.X) and (lcy >= zp.Y and lcy <= zp.Y + zs.Y)
end

function startAutoPuzzle()
    if qteConnection then qteConnection:Disconnect(); qteConnection = nil end
    if lineConnection then lineConnection:Disconnect(); lineConnection = nil end
    clickedZones = {}
    qteConnection = RunService.Heartbeat:Connect(function()
        if not autoPuzzleEnabled then
            if qteConnection then qteConnection:Disconnect(); qteConnection = nil end
            return
        end
        local qte = getQTE()
        if not qte then return end
        local line = qte:FindFirstChild("Line")
        if not line then return end
        local zones = getVisibleZones(qte)
        if #zones == 0 then clickedZones = {}; return end
        for _, z in ipairs(zones) do
            if z.Visible == true and isLineOverZone(line, z) then
                doClick()
                task.wait(0.1)
                if z.Visible == false then clickedZones[z] = true end
                task.wait(0.05)
                break
            end
        end
    end)
end

function stopAutoPuzzle()
    autoPuzzleEnabled = false
    if qteConnection then qteConnection:Disconnect(); qteConnection = nil end
    if lineConnection then lineConnection:Disconnect(); lineConnection = nil end
    clickedZones = {}
end

-- ===== ABILITY WARNER =====
function getShockwavePart(projectile)
    if not projectile then return nil end
    local p = projectile:FindFirstChild("Shockwave") or projectile:FindFirstChild("shockwave")
    if p and p:IsA("BasePart") then return p end
    if projectile:IsA("BasePart") then return projectile end
    for _, c in ipairs(projectile:GetChildren()) do
        if c:IsA("BasePart") then return c end
    end
    return nil
end

function getProjectileDirection(projectile, part)
    if part then
        local look = part.CFrame.LookVector
        if look.Magnitude > 0.1 then return look end
        local v = part:FindFirstChild("BodyVelocity")
        if v and v:IsA("BodyVelocity") and v.Velocity.Magnitude > 0.1 then return v.Velocity.Unit end
    end
    return nil
end

function createWarnerLineParts(folder, startPos, dir, length, color, transparency)
    local parts = {}
    local endPos = startPos + (dir * length)
    local yPos = startPos.Y
    local linePart = Instance.new("Part")
    linePart.Name = "AbilityWarningLine"
    linePart.Anchored = true
    linePart.CanCollide = false
    linePart.Material = Enum.Material.Neon
    linePart.Color = color
    linePart.Transparency = transparency
    linePart.Size = Vector3.new(0.3, 0.1, length)
    local mid = Vector3.new((startPos.X + endPos.X) / 2, yPos, (startPos.Z + endPos.Z) / 2)
    linePart.CFrame = CFrame.lookAt(mid, endPos)
    linePart.Parent = folder
    table.insert(parts, linePart)
    local glow = linePart:Clone()
    glow.Name = "AbilityWarningGlow"
    glow.Size = Vector3.new(1, 0.4, length)
    glow.Transparency = 0.7
    glow.Parent = folder
    table.insert(parts, glow)
    local arrow = Instance.new("Part")
    arrow.Name = "AbilityWarningArrow"
    arrow.Anchored = true
    arrow.CanCollide = false
    arrow.Material = Enum.Material.Neon
    arrow.Color = color
    arrow.Transparency = 0.3
    arrow.Shape = Enum.PartType.Cylinder
    arrow.Size = Vector3.new(1, 0.5, 1)
    arrow.Position = Vector3.new(endPos.X, yPos, endPos.Z)
    arrow.CFrame = CFrame.lookAt(Vector3.new(endPos.X, yPos, endPos.Z), Vector3.new(endPos.X + dir.X, yPos, endPos.Z + dir.Z))
    arrow.Parent = folder
    table.insert(parts, arrow)
    local ring = Instance.new("Part")
    ring.Name = "AbilityWarningRing"
    ring.Anchored = true
    ring.CanCollide = false
    ring.Material = Enum.Material.Neon
    ring.Color = color
    ring.Transparency = 0.4
    ring.Shape = Enum.PartType.Cylinder
    ring.Size = Vector3.new(2, 0.2, 2)
    ring.Position = Vector3.new(startPos.X, yPos, startPos.Z)
    ring.CFrame = CFrame.lookAt(Vector3.new(startPos.X, yPos, startPos.Z), Vector3.new(startPos.X + dir.X, yPos, startPos.Z + dir.Z))
    ring.Parent = folder
    table.insert(parts, ring)
    local n = math.floor(length / 10)
    for i = 1, n do
        local mp = startPos + (dir * (i * 10))
        local mk = Instance.new("Part")
        mk.Name = "AbilityWarningMarker"
        mk.Anchored = true
        mk.CanCollide = false
        mk.Material = Enum.Material.Neon
        mk.Color = color
        mk.Transparency = 0.5
        mk.Shape = Enum.PartType.Cylinder
        mk.Size = Vector3.new(0.5, 0.1, 0.5)
        mk.Position = Vector3.new(mp.X, yPos, mp.Z)
        mk.Parent = folder
        table.insert(parts, mk)
    end
    return parts
end

function updateWarnerLineParts(parts, startPos, dir, length, color, transparency)
    local endPos = startPos + (dir * length)
    local yPos = startPos.Y
    local mid = Vector3.new((startPos.X + endPos.X) / 2, yPos, (startPos.Z + endPos.Z) / 2)
    if parts[1] then parts[1].CFrame = CFrame.lookAt(mid, endPos) end
    if parts[2] then parts[2].CFrame = CFrame.lookAt(mid, endPos) end
    if parts[3] then
        parts[3].Position = Vector3.new(endPos.X, yPos, endPos.Z)
        parts[3].CFrame = CFrame.lookAt(Vector3.new(endPos.X, yPos, endPos.Z), Vector3.new(endPos.X + dir.X, yPos, endPos.Z + dir.Z))
    end
    if parts[4] then
        parts[4].Position = Vector3.new(startPos.X, yPos, startPos.Z)
        parts[4].CFrame = CFrame.lookAt(Vector3.new(startPos.X, yPos, startPos.Z), Vector3.new(startPos.X + dir.X, yPos, startPos.Z + dir.Z))
    end
    local n = math.floor(length / 10)
    for i = 1, n do
        if parts[4 + i] then
            local mp = startPos + (dir * (i * 10))
            parts[4 + i].Position = Vector3.new(mp.X, yPos, mp.Z)
        end
    end
end

function destroyWarnerLineData(d)
    if not d then return end
    if d.updateConnection then d.updateConnection:Disconnect(); d.updateConnection = nil end
    if d.cleanupConnection then d.cleanupConnection:Disconnect(); d.cleanupConnection = nil end
    if d.folder then pcall(function() d.folder:Destroy() end); d.folder = nil end
    d.isActive = false
end

function drawWarnerLineOnKiller(character, abilityName, duration)
    if not abilityWarnerEnabled then return end
    if not character or not character.Parent then return end
    if abilityWarnerKillerLineMap[character] then
        local ex = abilityWarnerKillerLineMap[character]
        destroyWarnerLineData(ex)
        for i, v in ipairs(abilityWarnerActiveLines) do
            if v == ex then table.remove(abilityWarnerActiveLines, i); break end
        end
        abilityWarnerKillerLineMap[character] = nil
    end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local folder = Instance.new("Folder")
    folder.Name = "AbilityWarning_Killer_" .. abilityName .. "_" .. tick()
    folder.Parent = Workspace
    local color = (abilityName == "Entanglement") and Color3.fromRGB(255, 165, 0) or Color3.fromRGB(255, 0, 0)
    local dir = hrp.CFrame.LookVector
    local sp = hrp.Position
    local parts = createWarnerLineParts(folder, sp, dir, abilityWarnerLineLength, color, abilityWarnerLineTransparency)
    local data = { folder = folder, parts = parts, character = character, hrp = hrp, abilityName = abilityName, isProjectile = false, duration = duration, isActive = true }
    data.updateConnection = RunService.Heartbeat:Connect(function()
        if not abilityWarnerEnabled or not data.isActive then return end
        if not character or not character.Parent or not hrp or not hrp.Parent then destroyWarnerLineData(data); return end
        updateWarnerLineParts(parts, hrp.Position, hrp.CFrame.LookVector, abilityWarnerLineLength, color, abilityWarnerLineTransparency)
    end)
    data.cleanupConnection = character.AncestryChanged:Connect(function()
        if not character or not character.Parent then destroyWarnerLineData(data) end
    end)
    table.insert(abilityWarnerActiveLines, data)
    abilityWarnerKillerLineMap[character] = data
    local fd = (abilityName == "Entanglement") and abilityWarnerEntanglementDuration or duration
    task.delay(fd, function()
        if data and data.isActive then
            data.isActive = false
            for i, v in ipairs(abilityWarnerActiveLines) do
                if v == data then table.remove(abilityWarnerActiveLines, i); break end
            end
            if abilityWarnerKillerLineMap[character] == data then abilityWarnerKillerLineMap[character] = nil end
            destroyWarnerLineData(data)
        end
    end)
    return data
end

function drawWarnerLineOnProjectile(projectile, abilityName, duration)
    if not abilityWarnerEnabled then return end
    if not projectile or not projectile.Parent then return end
    local pp = getShockwavePart(projectile)
    if not pp then return end
    local folder = Instance.new("Folder")
    folder.Name = "AbilityWarning_Projectile_" .. abilityName .. "_" .. tick()
    folder.Parent = Workspace
    local color = Color3.fromRGB(255, 0, 0)
    local sp = pp.Position
    local dir = getProjectileDirection(projectile, pp) or Vector3.new(1, 0, 0)
    if dir.Magnitude < 0.1 then dir = Vector3.new(1, 0, 0) end
    local parts = createWarnerLineParts(folder, sp, dir, abilityWarnerLineLength, color, abilityWarnerLineTransparency)
    local data = { folder = folder, parts = parts, projectile = projectile, projectilePart = pp, abilityName = abilityName, isProjectile = true, duration = duration, isActive = true }
    data.updateConnection = RunService.Heartbeat:Connect(function()
        if not abilityWarnerEnabled or not data.isActive then return end
        if not projectile or not projectile.Parent then destroyWarnerLineData(data); return end
        local cp = getShockwavePart(projectile)
        if not cp or not cp.Parent then return end
        local nd = getProjectileDirection(projectile, cp)
        if not nd or nd.Magnitude < 0.1 then nd = cp.CFrame.LookVector end
        if nd.Magnitude < 0.1 then nd = Vector3.new(1, 0, 0) end
        updateWarnerLineParts(parts, cp.Position, nd, abilityWarnerLineLength, color, abilityWarnerLineTransparency)
    end)
    data.cleanupConnection = projectile.AncestryChanged:Connect(function()
        if not projectile or not projectile.Parent then destroyWarnerLineData(data) end
    end)
    table.insert(abilityWarnerActiveLines, data)
    task.delay(duration or abilityWarnerMassInfectionDuration, function()
        if data and data.isActive then
            data.isActive = false
            for i, v in ipairs(abilityWarnerActiveLines) do
                if v == data then table.remove(abilityWarnerActiveLines, i); break end
            end
            destroyWarnerLineData(data)
        end
    end)
    return data
end

function removeWarnerKillerLine(killer)
    if not killer then return end
    if abilityWarnerKillerLineMap[killer] then
        local d = abilityWarnerKillerLineMap[killer]
        if d and d.isActive then
            d.isActive = false
            for i, v in ipairs(abilityWarnerActiveLines) do
                if v == d then table.remove(abilityWarnerActiveLines, i); break end
            end
            destroyWarnerLineData(d)
            abilityWarnerKillerLineMap[killer] = nil
        end
    end
end

    for _, c in pairs(detectionCircles) do safeDestroyCircle(c) end

                    end)
                end
            end
        end
    end
end

function updateStunImmuneIndicator(killer)

function scanAllKillersForStunImmune()

    end
end

function startStunImmuneLoop()
    if stunImmuneLoop then task.cancel(stunImmuneLoop); stunImmuneLoop = nil end
    if not stunImmuneEnabled then return end
    stunImmuneLoop = task.spawn(function()
        while stunImmuneEnabled do
            task.wait(0.5)
            scanAllKillersForStunImmune()
        end
    end)
end

function setupStunImmuneListeners()
    if not KillersFolder then return end
    KillersFolder.ChildAdded:Connect(function(k)
        task.wait(0.5)
        if stunImmuneEnabled then
            updateStunImmuneIndicator(k)
            local hrp = k:FindFirstChild("HumanoidRootPart")
            if hrp then
                hrp.ChildAdded:Connect(function(c)
                    if c.Name == "InvincibleFX" or c.Name == "StunProtection" then updateStunImmuneIndicator(k) end
                end)
                hrp.ChildRemoved:Connect(function(c)
                    if c.Name == "InvincibleFX" or c.Name == "StunProtection" then updateStunImmuneIndicator(k) end
                end)
            end
        end
    end)
    KillersFolder.ChildRemoved:Connect(function(k)
        local hrp = k:FindFirstChild("HumanoidRootPart")
        if hrp then
            local bb = hrp:FindFirstChild("StunImmuneBillboard")
            if bb then bb:Destroy() end
        end
    end)
end

function setupInputHook()
    UserInputService.InputBegan:Connect(function(input, gp)
        if gp then return end
        local myTeam, myCharacter = getMyCharacterInfo()
        local isShedletsky = (myTeam == "Survivor" and myCharacter == "Shedletsky")
        local isChance = (myTeam == "Survivor" and myCharacter == "Chance")
        local isTwoTime = (myTeam == "Survivor" and myCharacter == "TwoTime")
        local isElliot = (myTeam == "Survivor" and myCharacter == "Elliot")

        local isJohnDoe = (myTeam == "Killer" and myCharacter == "JohnDoe")
        local is1x1x1x1 = (myTeam == "Killer" and myCharacter == "1x1x1x1")
        local isSixer = (myTeam == "Killer" and myCharacter == "Sixer")
        local isc00lkidd = (myTeam == "Killer" and myCharacter == "c00lkidd")
        local isNoli = (myTeam == "Killer" and myCharacter == "Noli")
        local isAzureChar = (myTeam == "Killer" and myCharacter == "Azure")

        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            if not aimbotEnabled then return end
            if myTeam == "Killer" then
                local ac = lp.PlayerGui.MainUI.AbilityContainer
                if ac then
                    local cd = nil
                    if myCharacter == "Slasher" then
                        local s = ac:FindFirstChild("Slash"); if s then cd = s:FindFirstChild("CooldownTime") end
                    elseif myCharacter == "c00lkidd" then
                        local p = ac:FindFirstChild("Punch"); if p then cd = p:FindFirstChild("CooldownTime") end
                    elseif myCharacter == "JohnDoe" then
                        local s = ac:FindFirstChild("Slash"); if s then cd = s:FindFirstChild("CooldownTime") end
                    elseif myCharacter == "Noli" then
                        local s = ac:FindFirstChild("Stab"); if s then cd = s:FindFirstChild("CooldownTime") end
                    elseif myCharacter == "1x1x1x1" then
                        local s = ac:FindFirstChild("Slash"); if s then cd = s:FindFirstChild("CooldownTime") end
                    elseif myCharacter == "Sixer" then
                        local cs = ac:FindFirstChild("Carving Slash"); if cs then cd = cs:FindFirstChild("CooldownTime") end
                    elseif isAzureChar then
                        local s = ac:FindFirstChild("Slash"); if s then cd = s:FindFirstChild("CooldownTime") end
                    end
                    if cd then
                        local t = cd.ContentText
                        if t ~= "" and t ~= "0" and t ~= "0.0" and t ~= "0.1" then return end
                    end
                end
            end
            if myCharacter == "Sixer" then
                if getSixerCarvingSlashCharges() >= 1 then startGeneralAimbot(2.0) else startGeneralAimbot(0.8) end
            elseif myTeam == "Killer" then
                startGeneralAimbot(0.65)
            end
        end
        if input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == Enum.KeyCode.Q then
            if not aimbotEnabled then return end
            if isAzureChar then
                if getAzureForm() == "Beast" then
                    if not isAzureBeastQReady() then return end
                    local targets = getValidTargets()
                    if #targets == 0 then return end
                    startGeneralAimbot(azureBeastQDuration)
                end
                return
            end
            if isTwoTime then
                if not isTwoTimeDaggerReady() then SendNotif("TwoTime", "Dagger is on cooldown!", 1); return end
                local tk = getNearestKiller()
                if not tk then SendNotif("TwoTime", "No killer found!", 1); return end
                if isCrouching() then
                    SendNotif("TwoTime", "DASH STAB!", 1)
                    startDashStab(tk)
                else
                    aimbotActive = true
                    aimbotCurrentTarget = tk
                    if faceSameSideAsKiller then SendNotif("TwoTime", "Facing Same Side TRUE", 1) else SendNotif("TwoTime", "Facing Same Side FALSE", 1) end
                    startNormalStab(tk)
                end
            elseif isNoli then startNoliAimbot()
            elseif isSixer then startGeneralAimbot(1.45)
            elseif isJohnDoe then startGeneralAimbot(4.5)
            elseif isShedletsky then startGeneralAimbot(1.1)
            elseif is1x1x1x1 then startGeneralAimbot(2.1)
            elseif isSlasher then startGeneralAimbot(0.95)
            elseif isElliot then startGeneralAimbot(0.5) end
        end
        if input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == Enum.KeyCode.E then
            if not aimbotEnabled then return end
            if isAzureChar then
                if getAzureForm() == "Beast" then
                    local t = getValidTargets()
                    if #t == 0 then return end
                    startGeneralAimbot(azureBeastEDuration)
                end
                return
            end
            if isc00lkidd then startGeneralAimbot(0.6)
            elseif isSixer then startGeneralAimbot(1.0)
            elseif isChance then startGeneralAimbot(1.1)
            elseif is1x1x1x1 then startGeneralAimbot(1.2)
            elseif isSlasher then startGeneralAimbot(0.5)
            elseif isJaneDoe then startGeneralAimbot(0.9) end
        end
        if input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == Enum.KeyCode.R then
            if not aimbotEnabled then return end
            if isGuest1337 then startGeneralAimbot(0.95) end
        end
    end)
end
