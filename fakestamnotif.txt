local isExhausted = false

local lastTrickTime = 0

local exhaustedEffectConnection = nil

local isAbilityBlocking = false
local abilityBlockerConnections = {}

-- ROMAN NUMERAL CONVERSION
local romanToNumber = {
    ["I"] = 1, ["II"] = 2, ["III"] = 3, ["IV"] = 4, ["V"] = 5,
    ["VI"] = 6, ["VII"] = 7, ["VIII"] = 8, ["IX"] = 9, ["X"] = 10,
}

-- Exhausted helpers
local function getExhaustedLevel()
    local playerGui = localPlayer:FindFirstChild("PlayerGui")
    if not playerGui then return 0 end
    local mainUI = playerGui:FindFirstChild("MainUI")
    if not mainUI then return 0 end
    local statusContainer = mainUI:FindFirstChild("StatusContainer")
    if not statusContainer then return 0 end
    local exhausted = statusContainer:FindFirstChild("Exhausted")
    if not exhausted then return 0 end
    local title = exhausted:FindFirstChild("Title")
    if not title then return 1 end
    local text = title.Text
    if not text or text == "" then return 1 end
    local levelText = text:match("Exhausted%s+(.+)$")
    if not levelText then return 1 end
    if romanToNumber[levelText] then return romanToNumber[levelText] end
    local number = tonumber(levelText)
    if number then return number end
    return 1
end

local function isExhaustedActive()
    local playerGui = localPlayer:FindFirstChild("PlayerGui")
    if not playerGui then return false end
    local mainUI = playerGui:FindFirstChild("MainUI")
    if not mainUI then return false end
    local statusContainer = mainUI:FindFirstChild("StatusContainer")
    if not statusContainer then return false end
    return statusContainer:FindFirstChild("Exhausted") ~= nil
end

local function setupExhaustedUIListener()
    if exhaustedEffectConnection then
        exhaustedEffectConnection:Disconnect()
        exhaustedEffectConnection = nil
    end
    local function checkPlayerGui()
        local playerGui = localPlayer:FindFirstChild("PlayerGui")
        if not playerGui then return end
        local mainUI = playerGui:FindFirstChild("MainUI")
        if not mainUI then return end
        local statusContainer = mainUI:FindFirstChild("StatusContainer")
        if not statusContainer then return end
        if statusContainer:FindFirstChild("Exhausted") then
            isExhausted = true
            exhaustedLevel = getExhaustedLevel()
        end
        local addedConnection = statusContainer.ChildAdded:Connect(function(child)
            if child.Name == "Exhausted" then
                isExhausted = true
                exhaustedLevel = getExhaustedLevel()
            end
        end)
        table.insert(abilityBlockerConnections, addedConnection)
        local removedConnection = statusContainer.ChildRemoved:Connect(function(child)
            if child.Name == "Exhausted" then
                isExhausted = false
                exhaustedLevel = 0
            end
        end)
        table.insert(abilityBlockerConnections, removedConnection)
    end
    checkPlayerGui()
    local playerGui = localPlayer:FindFirstChild("PlayerGui")
    if playerGui then
        local mainUI = playerGui:FindFirstChild("MainUI")
        if not mainUI then
            local mainUIConnection = playerGui.ChildAdded:Connect(function(child)
                if child.Name == "MainUI" then
                    checkPlayerGui()
                    mainUIConnection:Disconnect()
                end
            end)
            table.insert(abilityBlockerConnections, mainUIConnection)
        end
    end
end

-- Ability detection
local function isUsingAbility()
    if not localCharacter then return false end
    local abilityIndicators = {
        "DrinkingCola", "DrinkingSlateskin", "EatingGhostburger",
        "StunningKiller", "PunchAbility", "TwoTimeStab", "jdaw",
        "ShootingGun", "UsingAbility", "AbilityActive",
    }
    for _, indicator in ipairs(abilityIndicators) do
        if localCharacter:FindFirstChild(indicator) then
            return true
        end
    end
    if speedMultipliers then
        for _, child in ipairs(speedMultipliers:GetChildren()) do
            if child.Name:match("Ability") or child.Name:match("Cola") or
               child.Name:match("Burger") or child.Name:match("Stun") or
               child.Name:match("Punch") or child.Name:match("Stab") or
               child.Name:match("Shooting") then
                return true
            end
        end
    end
    local characterTools = localCharacter:FindFirstChild("Tools")
    if characterTools then
        for _, tool in ipairs(characterTools:GetChildren()) do
            if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
                if tool.Name:match("Ability") or tool.Name:match("Cola") or
                   tool.Name:match("Burger") or tool.Name:match("Stun") or
                   tool.Name:match("Punch") or tool.Name:match("Stab") or
                   tool.Name:match("Gun") or tool.Name:match("Shoot") then
                    return true
                end
            end
        end
    end
    return false
end

-- Reset
local function resetReferences()
    speedMultipliers = nil
    sprintingValue = nil
    skateboardPart = nil
    veeronicaChar = nil
    isSkating = false
    wasSkating = false
    skatingDelay = 0
    isSprinting = false
    wasSprinting = false
    stamina = maxStamina
    timeUntilStaminaRecovers = 0
    staminaPenaltyActive = false
    isMoving = false
    isInKillerRange = false
    isAbilityBlocking = false
    isExhausted = false
    exhaustedLevel = 0
    lastTrickTime = 0
    for _, connection in pairs(abilityBlockerConnections) do
        if connection then
            connection:Disconnect()
        end
    end
    abilityBlockerConnections = {}
    if exhaustedEffectConnection then
        exhaustedEffectConnection:Disconnect()
        exhaustedEffectConnection = nil
    end
end

-- Ability blocker listeners
local function setupAbilityBlockerListeners()
    if not localCharacter then return end
    local addedConnection = localCharacter.ChildAdded:Connect(function(child)
        if child.Name:match("Ability") or child.Name:match("Cola") or
           child.Name:match("Burger") or child.Name:match("Stun") or
           child.Name:match("Punch") or child.Name:match("Stab") or
           child.Name:match("Shooting") or child.Name:match("jdaw") or
           child.Name:match("Using") or child.Name:match("Active") then
            isAbilityBlocking = true
            isSprinting = false
        end
    end)
    table.insert(abilityBlockerConnections, addedConnection)
    local removedConnection = localCharacter.ChildRemoved:Connect(function(child)
        if child.Name:match("Ability") or child.Name:match("Cola") or
           child.Name:match("Burger") or child.Name:match("Stun") or
           child.Name:match("Punch") or child.Name:match("Stab") or
           child.Name:match("Shooting") or child.Name:match("jdaw") or
           child.Name:match("Using") or child.Name:match("Active") then
            isAbilityBlocking = isUsingAbility()
        end
    end)
    table.insert(abilityBlockerConnections, removedConnection)
end

-- SpeedMultipliers init
local function initSpeedMultipliers()
    if not localCharacter then return false end
    speedMultipliers = localCharacter:FindFirstChild("SpeedMultipliers")
    if speedMultipliers then
        sprintingValue = speedMultipliers:FindFirstChild("Sprinting")
        return true
    end
    return false
end

-- Skateboard
local function getSkateboard()
    local survivors = Workspace:FindFirstChild("Players") and Workspace.Players:FindFirstChild("Survivors")
    if not survivors then return nil end
    veeronicaChar = survivors:FindFirstChild("Veeronica")
    if not veeronicaChar then return nil end
    if veeronicaChar:GetAttribute("Username") == localPlayer.Name then
        return veeronicaChar:FindFirstChild("Skateboard")
    end
    return nil
end

-- ===== TRICK DETECTION WITH ANTI-SPAM =====
local function setupTrickAudioDetection()
    task.spawn(function()
        while true do
            local survivors = Workspace:FindFirstChild("Players") and Workspace.Players:FindFirstChild("Survivors")
            if not survivors then
                task.wait(1)
                continue
            end
            veeronicaChar = survivors:FindFirstChild("Veeronica")
            if not veeronicaChar then
                task.wait(1)
                continue
            end
            if veeronicaChar:GetAttribute("Username") == localPlayer.Name then
                print("[Stamina] Veeronica detected! Setting up trick detection...")
                break
            else
                veeronicaChar:GetAttributeChangedSignal("Username"):Wait()
            end
        end

        local hrp = veeronicaChar:FindFirstChild("HumanoidRootPart")
        if not hrp then
            task.wait(0.5)
            setupTrickAudioDetection()
            return
        end

        print("[Stamina] Watching for trick sounds...")

        hrp.ChildAdded:Connect(function(child)
            if child:IsA("Sound") and child.SoundId == TRICK_AUDIO_ID then
                local now = tick()
                -- Check anti-spam
                if now - lastTrickTime >= TRICK_ANTI_SPAM then
                    lastTrickTime = now
                    local oldStamina = stamina
                    stamina = math.min(stamina + TRICK_STAMINA_GAIN, maxStamina)
                    print(string.format("[Stamina] TRICK! +%d stamina (%.2fs since last). Old: %d, New: %d", 
                        TRICK_STAMINA_GAIN, now - lastTrickTime, oldStamina, stamina))
                    
                    StaminaLabel.Text = string.format("Stamina: %d (TRICK!)", stamina)
                    StaminaLabel.TextColor3 = Color3.fromRGB(0, 255, 0)
                    task.delay(1, function()
                        StaminaLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
                    end)
                else
                    print(string.format("[Stamina] Trick ignored (anti-spam %.2fs remaining)", 
                        TRICK_ANTI_SPAM - (now - lastTrickTime)))
                end
            end
        end)
    end)
end

-- Inputs

        holdingShift = true
    end
end)
UserInputService.InputEnded:Connect(function(input, gp)
    if input.KeyCode == Enum.KeyCode.LeftShift then
        holdingShift = false
    end
end)

-- Movement
local function updateMovementCheck()
    if humanoidRootPart and humanoidRootPart:IsDescendantOf(workspace) then
        isMoving = humanoidRootPart.Velocity.Magnitude > VELOCITY_THRESHOLD
    end
end

-- Sprint
local function updateSprintingState()
    if not localCharacter then
        isSprinting = false
        return
    end
    isAbilityBlocking = isUsingAbility()
    if isAbilityBlocking then
        isSprinting = false
        return
    end
    if not speedMultipliers or not speedMultipliers.Parent then
        if not initSpeedMultipliers() then
            isSprinting = false
            return
        end
    end
    if sprintingValue then
        local currentValue = sprintingValue.Value
        if holdingShift and currentValue > 1.05 and isMoving then
            isSprinting = true
        else
            isSprinting = false
        end
    else
        isSprinting = false
    end
end

-- Killer range check
local function updateKillerRangeCheck()
    if not localCharacter or not humanoidRootPart then
        isInKillerRange = false
        return
    end
    
    local killerPosition = humanoidRootPart.Position
    local foundPlayer = false
    
    local playersFolder = Workspace:FindFirstChild("Players")
    if playersFolder then
        local survivorsFolder = playersFolder:FindFirstChild("Survivors")
        if survivorsFolder then
            for _, survivor in ipairs(survivorsFolder:GetChildren()) do
                local usernameAttr = survivor:GetAttribute("Username")
                if usernameAttr and usernameAttr ~= localPlayer.Name then
                    local targetRootPart = survivor:FindFirstChild("HumanoidRootPart")
                    if targetRootPart then
                        local distance = (killerPosition - targetRootPart.Position).Magnitude
                        if distance <= KILLER_CHECK_RANGE then
                            foundPlayer = true
                            break
                        end
                    end
                end
            end
        end
    end
    
    isInKillerRange = foundPlayer
end

local function canRegenStamina()
    return timeUntilStaminaRecovers <= 0 and skatingDelay <= 0 and not staminaPenaltyActive
end

-- Mode switch
local function setMode(mode)
    if mode == "Survivor" then
        maxStamina = 100
        stamina = maxStamina
        drainRate = 10
        regenRate = 20
        SurvivorButton.BackgroundColor3 = Color3.fromRGB(0, 100, 0)
        KillerButton.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
    elseif mode == "Killer" then
        maxStamina = 110
        stamina = maxStamina
        drainRate = 9.5
        regenRate = 21
        SurvivorButton.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
        KillerButton.BackgroundColor3 = Color3.fromRGB(100, 0, 0)
    end
    currentMode = mode
    isInKillerRange = false
    timeUntilStaminaRecovers = 0
    skatingDelay = 0
    staminaPenaltyActive = false
    isSprinting = false
    wasSprinting = false
    isSkating = false
    wasSkating = false
    isAbilityBlocking = false
    isExhausted = false
    exhaustedLevel = 0
end

SurvivorButton.MouseButton1Click:Connect(function()
    setMode("Survivor")
end)
KillerButton.MouseButton1Click:Connect(function()
    setMode("Killer")
end)

-- Skateboard update
local function updateSkateboardState()
    if not skateboardPart or not skateboardPart:IsDescendantOf(Workspace) then
        skateboardPart = getSkateboard()
    end
    if skateboardPart then
        isSkating = (skateboardPart.Transparency == 0)
    end
end

-- Initial setup
setupExhaustedUIListener()
setupTrickAudioDetection()

-- Main loop
RunService.Heartbeat:Connect(function(dt)
    if not localCharacter or not localCharacter.Parent then
        return
    end

    isExhausted = isExhaustedActive()
    if isExhausted then
        exhaustedLevel = getExhaustedLevel()
    else
        exhaustedLevel = 0
    end

    updateMovementCheck()
    updateSkateboardState()
    updateSprintingState()
    updateKillerRangeCheck()

    local currentDrainRate = drainRate
    local currentRegenRate = regenRate

    if isExhausted and exhaustedLevel > 0 then
        local drainMultiplier = EXHAUSTED_DRAIN_MULTIPLIER ^ exhaustedLevel
        local regenMultiplier = EXHAUSTED_REGEN_MULTIPLIER ^ exhaustedLevel
        currentDrainRate = currentDrainRate * drainMultiplier
        currentRegenRate = currentRegenRate * regenMultiplier
    end

    -- Determine if we should drain
    local shouldDrain = false
    if isSkating then
        shouldDrain = isMoving
    elseif currentMode == "Survivor" then
        shouldDrain = isSprinting and stamina > 0
    elseif currentMode == "Killer" then
        shouldDrain = isSprinting and isInKillerRange and stamina > 0
    end

    -- For killer, if out of range, treat as not sprinting for delay
    local effectiveSprinting = isSprinting
    if currentMode == "Killer" and isSprinting and not isInKillerRange then
        effectiveSprinting = false
    end

    wasSprinting = isSprinting
    wasSkating = isSkating

    -- Sprint delay (use effectiveSprinting)
    if effectiveSprinting then
        timeUntilStaminaRecovers = math.clamp(timeUntilStaminaRecovers + dt * 0.05, 0.2, 3)
    end
    if not effectiveSprinting and timeUntilStaminaRecovers > 0 then
        timeUntilStaminaRecovers = math.max(timeUntilStaminaRecovers - dt, 0)
    end

    -- Skating delay
    if isSkating then
        skatingDelay = math.clamp(skatingDelay + dt * 0.05, 0.2, 3)
    end
    if not isSkating and skatingDelay > 0 then
        skatingDelay = math.max(skatingDelay - dt, 0)
    end

    if shouldDrain and stamina > 0 then
        local finalDrainRate = currentDrainRate
        if isSkating then
            finalDrainRate = finalDrainRate * SKATING_DRAIN_MULTIPLIER
        end
        stamina = stamina - finalDrainRate * dt
        stamina = math.max(stamina, 0)

        if stamina <= 0 then
            stamina = 0
            timeUntilStaminaRecovers = 2
            skatingDelay = 2
            staminaPenaltyActive = true
            if localCharacter then
                localCharacter:SetAttribute("StaminaPenaltyActive", true)
            end
        end
    else
        if timeUntilStaminaRecovers <= 0 and skatingDelay <= 0 and staminaPenaltyActive then
            staminaPenaltyActive = false
            if localCharacter then
                localCharacter:SetAttribute("StaminaPenaltyActive", nil)
            end
        end

        if timeUntilStaminaRecovers <= 0 and skatingDelay <= 0 and not staminaPenaltyActive then
            if stamina < maxStamina then
                stamina = stamina + currentRegenRate * dt
                stamina = math.min(stamina, maxStamina)
            end
        end
    end

    -- Update labels
    local statusText = ""
    if isSkating then statusText = statusText .. " (SKATING)" end
    if currentMode == "Killer" then
        statusText = statusText .. (isInKillerRange and " (IN RANGE)" or " (OUT OF RANGE)")
    end
    if not isMoving then statusText = statusText .. " (STATIONARY)" end
    if isAbilityBlocking then statusText = statusText .. " (ABILITY)" end
    if isExhausted and exhaustedLevel > 0 then
        local romanDisplay = ""
        for roman, num in pairs(romanToNumber) do
            if num == exhaustedLevel then
                romanDisplay = roman
                break
            end
        end
        if romanDisplay == "" then 
            romanDisplay = tostring(exhaustedLevel) 
        end
        statusText = statusText .. string.format(" [EXHAUSTED %s]", romanDisplay)
    end
    if shouldDrain then
        statusText = statusText .. (stamina > 0 and " (SPRINTING)" or " (EXHAUSTED)")
    end
    if staminaPenaltyActive then statusText = statusText .. " (PENALTY)" end
    if stamina < maxStamina and not shouldDrain then
        if timeUntilStaminaRecovers > 0 or skatingDelay > 0 then
            local totalDelay = math.max(timeUntilStaminaRecovers, skatingDelay)
            statusText = statusText .. string.format(" (DELAY %.1fs)", totalDelay)
        elseif not staminaPenaltyActive and not isSkating and not effectiveSprinting then
            statusText = statusText .. " (REGENERATING)"
        end
    elseif stamina >= maxStamina then
        statusText = statusText .. " (FULL)"
    end

    StaminaLabel.Text = string.format("Stamina: %d%s", math.floor(stamina), statusText)

    -- Color
    if stamina < 20 or isExhausted then
        StaminaLabel.TextColor3 = Color3.fromRGB(255, 50, 50)
        StaminaLabel.BackgroundColor3 = isExhausted and Color3.fromRGB(50, 0, 30) or Color3.fromRGB(50, 0, 0)
    elseif stamina < 50 then
        StaminaLabel.TextColor3 = Color3.fromRGB(255, 200, 50)
        StaminaLabel.BackgroundColor3 = isExhausted and Color3.fromRGB(30, 10, 20) or Color3.fromRGB(30, 15, 0)
    else
        StaminaLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
        StaminaLabel.BackgroundColor3 = isExhausted and Color3.fromRGB(20, 0, 20) or Color3.fromRGB(0, 0, 0)
    end

    SurvivorButton.BackgroundTransparency = currentMode == "Survivor" and 0.1 or 0.3
    KillerButton.BackgroundTransparency = currentMode == "Killer" and 0.1 or 0.3
end)

-- Initial setup
setMode("Survivor")

if localCharacter then
    humanoid = localCharacter:FindFirstChild("Humanoid")
    humanoidRootPart = localCharacter:FindFirstChild("HumanoidRootPart")
    initSpeedMultipliers()
    setupExhaustedUIListener()
    setupTrickAudioDetection()
end
